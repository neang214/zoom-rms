### 3. `frontend/src/lib/currency.js` *(New shared module)*
Centralizes `KHR_RATE`, currency formatting, `orderTotal`, `toLocalDateStr`, HTML escaping, CSV formula-injection neutralization, and JPEG image normalization for both Menu and Stock uploads.

```js
export const KHR_RATE = 4100

export function formatUSD(amount) {
  const n = Number(amount) || 0
  return `$${n.toFixed(2)}`
}

export function formatKHR(amountUSD) {
  const n = Number(amountUSD) || 0
  return `៛${Math.round(n * KHR_RATE).toLocaleString()}`
}

export function orderTotal(order) {
  return (
    order?.order_items?.reduce(
      (sum, oi) => sum + Number(oi.unit_price || 0) * Number(oi.quantity || 0),
      0
    ) || 0
  )
}

export function toLocalDateStr(date = new Date()) {
  const d = date instanceof Date ? date : new Date(date)
  const y = d.getFullYear()
  const m = String(d.getMonth() + 1).padStart(2, "0")
  const day = String(d.getDate()).padStart(2, "0")
  return `${y}-${m}-${day}`
}

export function escapeHtml(str) {
  return String(str ?? "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;")
}

/**
 * Neutralizes spreadsheet formula injection (cells starting with =, +, -, @, tab, CR)
 * before quoting for CSV export.
 */
export function sanitizeCsvCell(val) {
  let s = String(val ?? "")
  if (/^[=+\-@\t\r]/.test(s)) {
    s = `'${s}`
  }
  return `"${s.replace(/"/g, '""')}"`
}

/**
 * Resizes and normalizes uploaded photos to JPEG so phone photos don't bloat uploads.
 */
export function normalizeImageToJpg(file, maxWidth = 800, quality = 0.85) {
  return new Promise((resolve) => {
    const img = new Image()
    const objectUrl = URL.createObjectURL(file)

    img.onload = () => {
      URL.revokeObjectURL(objectUrl)
      let { width, height } = img
      if (width > maxWidth) {
        height = Math.round((height * maxWidth) / width)
        width = maxWidth
      }
      const canvas = document.createElement("canvas")
      canvas.width = width
      canvas.height = height
      const ctx = canvas.getContext("2d")
      ctx.fillStyle = "#ffffff"
      ctx.fillRect(0, 0, width, height)
      ctx.drawImage(img, 0, 0, width, height)

      canvas.toBlob(
        (blob) => {
          if (!blob) {
            resolve(file)
            return
          }
          const cleanName = (file.name || "image").replace(/\.[^.]+$/, "") + ".jpg"
          resolve(new File([blob], cleanName, { type: "image/jpeg" }))
        },
        "image/jpeg",
        quality
      )
    }

    img.onerror = () => {
      URL.revokeObjectURL(objectUrl)
      resolve(file)
    }

    img.src = objectUrl
  })
}
```

### 5. `frontend/src/services/axios.js`
- Wires up `useLoadingBarStore` for smooth global request feedback.
- Stops sending `x-guest-session` on authenticated staff API calls.
- Triggers clean Zustand logout on staff `401` so expired sessions redirect to `/login` automatically.

```js
import axios from "axios"
import { useLoadingBarStore } from "../context/loadingBarContext"

export const GUEST_SESSION_KEY = "zoom_guest_session"

let onUnauthorized = null

export function setUnauthorizedHandler(handler) {
  onUnauthorized = handler
}

const clientApi = axios.create({
  baseURL: (import.meta.env.VITE_API_URL || "/api").replace(/\/$/, ""),
  withCredentials: true,
  headers: { "Content-Type": "application/json" },
})

clientApi.interceptors.request.use(
  (config) => {
    if (!config.silent) {
      useLoadingBarStore.getState().start()
      config._trackedByLoadingBar = true
    }

    const token = sessionStorage.getItem("rms_token")
    if (token) {
      config.headers["Authorization"] = `Bearer ${token}`
    } else {
      const guestSession = localStorage.getItem(GUEST_SESSION_KEY)
      if (guestSession) {
        config.headers["x-guest-session"] = guestSession
      }
    }

    return config
  },
  (error) => Promise.reject(error)
)

clientApi.interceptors.response.use(
  (response) => {
    if (response.config?._trackedByLoadingBar) {
      useLoadingBarStore.getState().stop()
    }
    return response
  },
  (error) => {
    if (error?.config?._trackedByLoadingBar) {
      useLoadingBarStore.getState().stop()
    }

    if (error?.response?.status === 401) {
      const hadStaffSession =
        Boolean(sessionStorage.getItem("rms_token")) ||
        sessionStorage.getItem("rms-tab-authenticated") === "true"

      sessionStorage.removeItem("rms_token")
      sessionStorage.removeItem("rms-tab-authenticated")
      sessionStorage.removeItem("rms-auth-storage")
      localStorage.removeItem("rms_token")

      if (hadStaffSession) {
        onUnauthorized?.()
      }
    }

    return Promise.reject(error)
  }
)

export default clientApi
```

### 6. `frontend/src/context/authContext.js`
Connects to `setUnauthorizedHandler` and properly tracks `error` state.

```js
import { create } from "zustand"
import { persist, createJSONStorage } from "zustand/middleware"
import api, { setUnauthorizedHandler } from "../services/axios"
import { resetSocket } from "../services/socket"

const TAB_SESSION_KEY = "rms-tab-authenticated"

export const authContext = create(
  persist(
    (set) => ({
      user: null,
      isLoading: false,
      error: null,

      checkAuth: async () => {
        const isTabAuthed = sessionStorage.getItem(TAB_SESSION_KEY) === "true"
        const token = sessionStorage.getItem("rms_token")

        if (!isTabAuthed || !token) {
          localStorage.removeItem("rms_token")
          set({ user: null, isLoading: false, error: null })
          return
        }

        set({ isLoading: true, error: null })
        try {
          const res = await api.get("/auth/me")
          const userData = res.data.user || res.data
          set({ user: userData, isLoading: false, error: null })
        } catch (error) {
          if (error.response?.status === 401 || error.response?.status === 403) {
            sessionStorage.removeItem(TAB_SESSION_KEY)
            sessionStorage.removeItem("rms_token")
            sessionStorage.removeItem("rms-auth-storage")
            localStorage.removeItem("rms_token")
            set({ user: null, isLoading: false })
            return
          }
          set({ isLoading: false })
        }
      },

      login: async (formData) => {
        set({ isLoading: true, error: null })
        try {
          const res = await api.post("/auth/login", formData)
          const userData = res.data.user || res.data
          if (res.data.token) {
            sessionStorage.setItem("rms_token", res.data.token)
          }
          sessionStorage.setItem(TAB_SESSION_KEY, "true")
          localStorage.removeItem("rms_token")
          resetSocket()
          set({ user: userData, isLoading: false, error: null })
          return userData
        } catch (error) {
          const msg =
            error?.response?.data?.message ||
            error?.response?.data?.error ||
            null
          set({ user: null, isLoading: false, error: msg })
          throw error
        }
      },

      logout: async () => {
        try {
          await api.post("/auth/logout")
        } catch (err) {
          console.error("Logout request error:", err)
        } finally {
          sessionStorage.removeItem(TAB_SESSION_KEY)
          sessionStorage.removeItem("rms_token")
          sessionStorage.removeItem("rms-auth-storage")
          localStorage.removeItem("rms_token")
          localStorage.removeItem("rms-auth-storage")
          resetSocket()
          set({ user: null, isLoading: false, error: null })
        }
      },
    }),
    {
      name: "rms-auth-storage",
      storage: createJSONStorage(() => sessionStorage),
      partialize: (state) => ({ user: state.user }),
    }
  )
)

setUnauthorizedHandler(() => {
  if (authContext.getState().user) {
    resetSocket()
    authContext.setState({ user: null, isLoading: false, error: null })
  }
})
```

### 7. `frontend/src/services/socket.js`
Fixes the uncleaned `disconnect` listener memory leaks and safely rebinds active hooks if `resetSocket()` runs.

```js
import { io } from "socket.io-client"
import { useEffect, useRef, useState } from "react"

const SOCKET_URL = (import.meta.env.VITE_SOCKET_URL || "").replace(/\/$/, "")

let socket = null
const resetSubscribers = new Set()

function getSocket() {
  if (!socket) {
    socket = io(SOCKET_URL, {
      withCredentials: true,
      transports: ["websocket", "polling"],
      autoConnect: true,
      reconnection: true,
      reconnectionDelay: 1000,
      reconnectionDelayMax: 5000,
    })
  }
  return socket
}

function useSocketEpoch() {
  const [epoch, setEpoch] = useState(0)
  useEffect(() => {
    const bump = () => setEpoch((n) => n + 1)
    resetSubscribers.add(bump)
    return () => {
      resetSubscribers.delete(bump)
    }
  }, [])
  return epoch
}

export function useSocketRole(role) {
  const epoch = useSocketEpoch()
  const [connected, setConnected] = useState(false)

  useEffect(() => {
    if (!role) return
    const s = getSocket()

    const onConnect = () => {
      s.emit("join_role", role)
      setConnected(true)
    }
    const onDisconnect = () => setConnected(false)

    if (s.connected) onConnect()
    s.on("connect", onConnect)
    s.on("disconnect", onDisconnect)

    return () => {
      s.off("connect", onConnect)
      s.off("disconnect", onDisconnect)
    }
  }, [role, epoch])

  return { socket: getSocket(), connected }
}

export function useSocketTable(tableId) {
  const epoch = useSocketEpoch()
  const [connected, setConnected] = useState(false)

  useEffect(() => {
    if (!tableId) return
    const s = getSocket()

    const onConnect = () => {
      s.emit("join_table", tableId)
      setConnected(true)
    }
    const onDisconnect = () => setConnected(false)

    if (s.connected) onConnect()
    s.on("connect", onConnect)
    s.on("disconnect", onDisconnect)

    return () => {
      s.emit("leave_table", tableId)
      s.off("connect", onConnect)
      s.off("disconnect", onDisconnect)
    }
  }, [tableId, epoch])

  return { socket: getSocket(), connected }
}

export function useSocketEvent(eventName, handler) {
  const epoch = useSocketEpoch()
  const handlerRef = useRef(handler)
  handlerRef.current = handler

  useEffect(() => {
    if (!eventName) return
    const s = getSocket()
    const wrapped = (...args) => handlerRef.current?.(...args)
    s.on(eventName, wrapped)
    return () => s.off(eventName, wrapped)
  }, [eventName, epoch])
}

export function resetSocket() {
  if (socket) {
    socket.disconnect()
    socket = null
  }
  resetSubscribers.forEach((fn) => fn())
}

export default getSocket
```

```jsx
                >
                  <AlertCircle size={15} className="shrink-0" />
                  <span>{displayError}</span>
                </div>
              )}

              <button
                type="submit"
                disabled={isLoading}
                className={cn(
                  "w-full h-11 rounded-xl font-semibold text-sm flex items-center justify-center gap-2 transition-all mt-2 shadow-xs cursor-pointer",
                  isLoading
                    ? "bg-[var(--color-primary)]/60 text-white/85 cursor-not-allowed"
                    : "bg-[var(--color-primary)] text-white hover:bg-[var(--color-primary-dark)] active:scale-[0.99]"

              >

                  <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
                ) : (
                  <>
                    <LogIn size={16} />
                    <span>{t("login.signInToSystem")}</span>
                  </>
                )}
              </button>
            </form>
          </div>
        </div>
      </div>
    </div>
  )
}
```

### 11. `frontend/src/App.jsx`
Adds **`React.lazy` + `<Suspense>` code splitting** so mobile guests only download the lightweight Guest bundle (instead of Admin, Reports, Stock, and `jsbarcode`), plus a refined branded splash screen.

```jsx
import { lazy, Suspense, useEffect, useState } from "react"
import { BrowserRouter, Routes, Route, Navigate, useNavigate, useLocation } from "react-router-dom"
import { ThemeProvider } from "next-themes"
import { Leaf } from "lucide-react"
import { authContext } from "./context/authContext"
import Layout from "./components/Layout"

const Login = lazy(() => import("./pages/Login"))

const Dashboard = lazy(() => import("./pages/admin/Dashboard"))
const Menu = lazy(() => import("./pages/admin/Menu"))
const Stock = lazy(() => import("./pages/admin/Stock"))
const Tables = lazy(() => import("./pages/admin/Tables"))
const Payments = lazy(() => import("./pages/admin/Payments"))
const Reports = lazy(() => import("./pages/admin/Reports"))
const Suppliers = lazy(() => import("./pages/admin/Suppliers"))
const UserManagement = lazy(() => import("./pages/admin/UserManagement"))
const OrderHistory = lazy(() => import("./pages/admin/OrderHistory"))

const CashierMenu = lazy(() => import("./pages/cashier/CashierMenu"))
const CashierOrders = lazy(() => import("./pages/cashier/CashierOrders"))
const CashierPaymentsReport = lazy(() => import("./pages/cashier/PaymentsReport"))
const CashierHistory = lazy(() => import("./pages/cashier/CashierHistory"))

const BaristaMenu = lazy(() => import("./pages/barista/BaristaMenu"))
const BaristaOrders = lazy(() => import("./pages/barista/BaristaOrders"))
const BaristaStock = lazy(() => import("./pages/barista/BaristaStock"))

const KitchenMenu = lazy(() => import("./pages/kitchen/KitchenMenu"))
const KitchenOrders = lazy(() => import("./pages/kitchen/KitchenOrders"))
const KitchenStock = lazy(() => import("./pages/kitchen/KitchenStock"))

const GuestLanding = lazy(() => import("./pages/guest/GuestLanding"))
const GuestOrder = lazy(() => import("./pages/guest/GuestOrder"))
const GuestOrderStatus = lazy(() => import("./pages/guest/GuestOrderStatus"))

const roleHome = {
  admin: "/admin",
  cashier: "/cashier/menu",
  kitchen: "/kitchen/menu",
  barista: "/barista/menu",
}

function PageFallback({ fullScreen = false }) {
  return (
    <div
      className={
        fullScreen
          ? "h-dvh w-screen flex flex-col items-center justify-center gap-4 bg-[var(--color-background)]"
          : "min-h-[40vh] w-full flex flex-col items-center justify-center gap-3"
      }
    >
      <div className="flex items-center gap-2.5">
        <div className="w-9 h-9 rounded-xl bg-[var(--color-primary)] text-white flex items-center justify-center shadow-xs">
          <Leaf size={17} className="rotate-12" />
        </div>
        <div>
          <div className="font-display font-bold text-xl text-[var(--color-text)] leading-none">
            ZOOM
          </div>
          <div className="text-[11px] text-[var(--color-muted)] mt-0.5">
            Garden Café &amp; Wine
          </div>
        </div>
      </div>
      <div className="w-5 h-5 border-2 border-[var(--color-primary)]/30 border-t-[var(--color-primary)] rounded-full animate-spin" />
    </div>
  )
}

function isAllowed(role, pathname) {
  const prefixMap = {
    admin: "/admin",
    cashier: "/cashier",
    kitchen: "/kitchen",
    barista: "/barista",
  }
  const prefix = prefixMap[role]
  if (!prefix) return false
  return pathname === prefix || pathname.startsWith(prefix + "/")
}

function RoleRedirector() {
  const { user } = authContext()
  const role = user?.role
  const navigate = useNavigate()
  const location = useLocation()

  useEffect(() => {
    if (!role) return
    if (!isAllowed(role, location.pathname)) {
      navigate(roleHome[role] || "/admin", { replace: true })
    }
  }, [role, location.pathname, navigate])

  return null
}

function ProtectedRoute({ role, allowedRoles, children }) {
  if (!allowedRoles.includes(role)) {
    return <Navigate to={roleHome[role] || "/admin"} replace />
  }
  return children
}

function StaffApp() {
  const { user, checkAuth } = authContext()
  const [initializing, setInitializing] = useState(true)

  useEffect(() => {
    Promise.resolve(checkAuth()).finally(() => setInitializing(false))
  }, [checkAuth])

  if (initializing) {
    return <PageFallback fullScreen />
  }

  if (!user) {
    return (
      <Suspense fallback={<PageFallback fullScreen />}>
        <Routes>
          <Route path="/login" element={<Login />} />
          <Route path="*" element={<Navigate to="/login" replace />} />
        </Routes>
      </Suspense>
    )
  }

  const role = user.role || "admin"

  return (
    <Layout>
      <RoleRedirector />
      <Suspense fallback={<PageFallback />}>
        <Routes>
          <Route path="/login" element={<Navigate to={roleHome[role] || "/admin"} replace />} />

          {/* Admin */}
          <Route path="/admin" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><Dashboard /></ProtectedRoute>} />
          <Route path="/admin/menu" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><Menu /></ProtectedRoute>} />
          <Route path="/admin/menus" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><Menu /></ProtectedRoute>} />
          <Route path="/admin/stock" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><Stock /></ProtectedRoute>} />
          <Route path="/admin/tables" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><Tables /></ProtectedRoute>} />
          <Route path="/admin/payments" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><Payments /></ProtectedRoute>} />
          <Route path="/admin/reports" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><Reports /></ProtectedRoute>} />
          <Route path="/admin/suppliers" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><Suppliers /></ProtectedRoute>} />
          <Route path="/admin/users" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><UserManagement /></ProtectedRoute>} />
          <Route path="/admin/history" element={<ProtectedRoute role={role} allowedRoles={["admin"]}><OrderHistory /></ProtectedRoute>} />

          {/* Cashier */}
          <Route path="/cashier/menu" element={<ProtectedRoute role={role} allowedRoles={["cashier"]}><CashierMenu /></ProtectedRoute>} />
          <Route path="/cashier/orders" element={<ProtectedRoute role={role} allowedRoles={["cashier"]}><CashierOrders /></ProtectedRoute>} />
          <Route path="/cashier/payments" element={<ProtectedRoute role={role} allowedRoles={["cashier"]}><CashierPaymentsReport /></ProtectedRoute>} />
          <Route path="/cashier/history" element={<ProtectedRoute role={role} allowedRoles={["cashier"]}><CashierHistory /></ProtectedRoute>} />

          {/* Kitchen */}
          <Route path="/kitchen/menu" element={<ProtectedRoute role={role} allowedRoles={["kitchen"]}><KitchenMenu /></ProtectedRoute>} />
          <Route path="/kitchen/orders" element={<ProtectedRoute role={role} allowedRoles={["kitchen"]}><KitchenOrders /></ProtectedRoute>} />
          <Route path="/kitchen/stock" element={<ProtectedRoute role={role} allowedRoles={["kitchen"]}><KitchenStock /></ProtectedRoute>} />

          {/* Barista */}
          <Route path="/barista/menu" element={<ProtectedRoute role={role} allowedRoles={["barista"]}><BaristaMenu /></ProtectedRoute>} />
          <Route path="/barista/orders" element={<ProtectedRoute role={role} allowedRoles={["barista"]}><BaristaOrders /></ProtectedRoute>} />
          <Route path="/barista/stock" element={<ProtectedRoute role={role} allowedRoles={["barista"]}><BaristaStock /></ProtectedRoute>} />

          <Route path="*" element={<Navigate to={roleHome[role] || "/admin"} replace />} />
        </Routes>
      </Suspense>
    </Layout>
  )
}

export default function App() {
  return (
    <ThemeProvider attribute="class" defaultTheme="light" enableSystem={false}>
      <BrowserRouter>
        <Suspense fallback={<PageFallback fullScreen />}>
          <Routes>
            <Route path="/" element={<GuestLanding />} />
            <Route path="/order" element={<GuestOrder />} />
            <Route path="/order/status" element={<GuestOrderStatus />} />
            <Route path="/*" element={<StaffApp />} />
          </Routes>
        </Suspense>
      </BrowserRouter>
    </ThemeProvider>
  )
}
```
