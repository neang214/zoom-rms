```js
import { create } from "zustand"
import clientApi from "../services/axios"

export const useDashboardStore = create((set, get) => {
  const runTracked = async (url, key, fallbackErr) => {
    set((s) => ({
      pendingCount: s.pendingCount + 1,
      isLoading: true,
      error: null,
    }))
    try {
      const res = await clientApi.get(url)
      set((s) => {
        const next = Math.max(0, s.pendingCount - 1)
        return { [key]: res.data, pendingCount: next, isLoading: next > 0 }
      })
      return res.data
    } catch (error) {
      set((s) => {
        const next = Math.max(0, s.pendingCount - 1)
        return {
          error: error.message || fallbackErr,
          pendingCount: next,
          isLoading: next > 0,
        }
      })
      throw error
    }
  }

  return {
    dailyRevenue: null,
    weeklyRevenue: null,
    monthlyRevenue: null,
    topSellingItems: [],
    peakHours: [],

    pendingCount: 0,
    isLoading: false,
    error: null,

    fetchDailyRevenue: () =>
      runTracked("/analytics/revenue/daily", "dailyRevenue", "Failed to fetch daily revenue"),

    fetchWeeklyRevenue: () =>
      runTracked("/analytics/revenue/weekly", "weeklyRevenue", "Failed to fetch weekly revenue"),

    fetchMonthlyRevenue: () =>
      runTracked("/analytics/revenue/monthly", "monthlyRevenue", "Failed to fetch monthly revenue"),

    fetchTopSelling: () =>
      runTracked("/analytics/menu/top-selling", "topSellingItems", "Failed to fetch top selling items"),

    fetchPeakHours: () =>
      runTracked("/analytics/orders/peak-hours", "peakHours", "Failed to fetch peak hours"),

    fetchAll: async () => {
      const s = get()
      await Promise.allSettled([
        s.fetchDailyRevenue(),
        s.fetchWeeklyRevenue(),
        s.fetchMonthlyRevenue(),
        s.fetchTopSelling(),
        s.fetchPeakHours(),
      ])
    },
  }
})
```

```jsx
function formatHourRange(hourNum) {
  const fmt = (h) => {
    const normalized = ((h % 24) + 24) % 24
    const suffix = normalized >= 12 ? "PM" : "AM"
    const hr12 = normalized % 12 === 0 ? 12 : normalized % 12
    return `${hr12}:00 ${suffix}`
  }
  const h = Number(hourNum)
  return `${fmt(h)} – ${fmt(h + 1)}`
}
```

```jsx
import { useEffect, useCallback, useRef } from "react"
import { RefreshCw } from "lucide-react"
import { useDashboardStore } from "../../context/dashboardContext"
import { usePageTitle } from "../../hooks/usePageTitle"
import { useLang } from "@/i18n/LanguageContext"
import { useSocketRole, useSocketEvent } from "../../services/socket"
import { RevenueStats, TopSellingTable, PeakHoursCard } from "@/components/admin/DashboardParts"

export default function Dashboard() {
  const { t } = useLang()
  usePageTitle(t("dashboard.title"))
  const {
    dailyRevenue,
    weeklyRevenue,
    monthlyRevenue,
    topSellingItems,
    peakHours,
    isLoading,
    fetchAll,
  } = useDashboardStore()

  const refreshTimerRef = useRef(null)

  const refreshNow = useCallback(() => {
    fetchAll()
  }, [fetchAll])

  const scheduleRefresh = useCallback(() => {
    if (refreshTimerRef.current) clearTimeout(refreshTimerRef.current)
    refreshTimerRef.current = setTimeout(() => {
      refreshTimerRef.current = null
      fetchAll()
    }, 400)
  }, [fetchAll])

  useEffect(() => {
    refreshNow()
    return () => {
      if (refreshTimerRef.current) clearTimeout(refreshTimerRef.current)
    }
  }, [refreshNow])

  useSocketRole("admin")
  useSocketEvent("payment:completed", scheduleRefresh)
  useSocketEvent("order:new", scheduleRefresh)
  useSocketEvent("order:status_changed", scheduleRefresh)

  return (
    <div className="space-y-6">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
        <div>
          <div className="flex items-center gap-2.5">
```

```jsx
            </div>
          )}
        </div>
      )}
    </div>
  )
}

export default function OrderHistoryView({ canPrint = false }) {
  const { t, lang } = useLang()
  usePageTitle(t("nav.history"))
  const { fetchOrderHistory } = useOrderStore()
  const { tables, fetchTables } = useTableStore()

  const todayStr = toLocalDateStr()
  const [date, setDate] = useState(todayStr)
  const [tableId, setTableId] = useState("")
  const [search, setSearch] = useState("")
  const [data, setData] = useState(null)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState(null)

  useEffect(() => {
    fetchTables().catch(() => {})
  }, [fetchTables])

  useEffect(() => {
    let cancelled = false
    setLoading(true)
    setError(null)
    fetchOrderHistory(date, tableId || undefined)
      .then((res) => {
        if (!cancelled) setData(res)
      })
      .catch(() => {
        if (!cancelled) setError(t("cashier.submitFailed"))
      })
      .finally(() => {
        if (!cancelled) setLoading(false)
      })
    return () => {
      cancelled = true
    }
  }, [date, tableId, fetchOrderHistory, t])

  const prevDay = () => {
    const d = new Date(`${date}T12:00:00`)
    d.setDate(d.getDate() - 1)
    setDate(toLocalDateStr(d))
  }

  const nextDay = () => {
    const d = new Date(`${date}T12:00:00`)
    d.setDate(d.getDate() + 1)
    const nextStr = toLocalDateStr(d)
    if (nextStr <= todayStr) setDate(nextStr)
  }

  const isToday = date === todayStr

  const displayDate = new Date(`${date}T12:00:00`).toLocaleDateString(
    lang === "km" ? "km-KH" : undefined,
    { weekday: "long", year: "numeric", month: "long", day: "numeric" }
  )

  const tableLabel = tableId
    ? tables.find((tb) => String(tb.table_id) === String(tableId))?.table_number
    : null

  const searchQuery = search.trim().toLowerCase()
  const searchDigits = searchQuery.replace(/\D/g, "")
  const searchAsNumber = searchDigits ? parseInt(searchDigits, 10) : null

  const filteredOrders = useMemo(() => {
    const list = data?.orders || []
    if (!searchQuery) return list
    return list.filter((o) => {
      const orderIdStr = String(o.order_id)
      const padded = orderIdStr.padStart(4, "0")
      if (
        (searchAsNumber !== null && o.order_id === searchAsNumber) ||
        orderIdStr.includes(searchQuery) ||
        padded.includes(searchQuery) ||
        `#${orderIdStr}`.includes(searchQuery) ||
        `#${padded}`.includes(searchQuery) ||
        `ord-${padded}`.includes(searchQuery) ||
        `ord-${orderIdStr}`.includes(searchQuery)
      ) {
        return true
      }
      if (String(o.table?.table_number ?? "").toLowerCase().includes(searchQuery)) return true
      if (o.note && o.note.toLowerCase().includes(searchQuery)) return true
      if (o.order_items?.some((oi) => menuItemMatchesSearch(oi.menu_item, searchQuery))) return true
      if (o.payments?.some((p) => p.method?.method_name?.toLowerCase().includes(searchQuery))) {
        return true
      }
      return false
    })
  }, [data?.orders, searchQuery, searchAsNumber])

  const canPrintDaily = canPrint && !loading && data && data.orders?.length > 0

  return (
    <div className="space-y-4">
      {/* Header + Date Picker */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
        <div>
          <h1 className="font-display text-2xl font-bold text-[var(--color-text)] flex items-center gap-2.5">
            <Clock size={21} className="text-[var(--color-primary)]" />
            <span>{t("nav.history")}</span>
          </h1>
          <p className="text-xs sm:text-sm text-[var(--color-muted)] mt-0.5">
            {t("history.dailySubtitle")}
          </p>
        </div>

        <div className="inline-flex items-center gap-1.5 bg-[var(--color-surface)] border border-[var(--color-border)] rounded-xl px-2.5 py-1.5 self-start sm:self-auto">
          <button
            type="button"
            onClick={prevDay}
            aria-label="Previous day"
            className="p-1.5 rounded-lg hover:bg-[var(--color-surface-subtle)] text-[var(--color-muted)] hover:text-[var(--color-text)] cursor-pointer"
          >
            <ChevronLeft size={15} />
          </button>
          <input
            type="date"
            value={date}
            max={todayStr}
            onChange={(e) => e.target.value && setDate(e.target.value)}
            className="text-xs sm:text-sm font-semibold text-[var(--color-text)] bg-transparent outline-none cursor-pointer"
          />
          <button
            type="button"
            onClick={nextDay}
            disabled={isToday}
            aria-label="Next day"
            className={cn(
              "p-1.5 rounded-lg text-[var(--color-muted)] disabled:opacity-35",
              !isToday && "hover:bg-[var(--color-surface-subtle)] hover:text-[var(--color-text)] cursor-pointer"
            )}
          >
            <ChevronRight size={15} />
          </button>
        </div>
      </div>

      {/* Filter Bar */}
      <div className="flex items-center justify-between gap-2.5 flex-wrap">
        <div className="flex items-center gap-2 flex-1 min-w-[260px]">
          <div className="relative flex-1 max-w-md">
            <Search
              size={15}
              className="absolute left-3.5 top-1/2 -translate-y-1/2 text-[var(--color-muted)] pointer-events-none"
            />
            <input
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              placeholder={t("cashier.searchPlaceholder")}
              className="w-full h-10 pl-10 pr-9 rounded-xl border border-[var(--color-border)] bg-[var(--color-surface)] text-sm text-[var(--color-text)] outline-none focus:ring-2 focus:ring-[var(--color-primary)]/25 focus:border-[var(--color-primary)]"
            />
            {search && (
              <button
                type="button"
                onClick={() => setSearch("")}
                aria-label="Clear search"
                className="absolute right-2.5 top-1/2 -translate-y-1/2 p-1 rounded-full text-[var(--color-muted)] hover:bg-[var(--color-surface-subtle)] cursor-pointer"
              >
                <X size={14} />
              </button>
            )}
          </div>

          <select
            value={tableId}
            onChange={(e) => setTableId(e.target.value)}
            aria-label={t("common.table")}
            className="h-10 text-xs sm:text-sm font-semibold text-[var(--color-text)] bg-[var(--color-surface)] border border-[var(--color-border)] rounded-xl px-3 outline-none focus:border-[var(--color-primary)] cursor-pointer"
          >
            <option value="">{t("tables.allTables")}</option>
            {tables.map((tb) => (
              <option key={tb.table_id} value={tb.table_id}>
                {tb.table_number}
              </option>
            ))}
          </select>
        </div>

        {canPrint && (
          <button
            type="button"
            onClick={() =>
              canPrintDaily &&
              printDailyReport({ date, displayDate, tableLabel, data })
            }
            disabled={!canPrintDaily}
            className="flex items-center gap-1.5 text-xs sm:text-sm font-semibold px-4 h-10 rounded-xl bg-[var(--color-surface)] border border-[var(--color-border)] text-[var(--color-text-secondary)] hover:bg-[var(--color-primary-muted)] hover:text-[var(--color-primary)] hover:border-[var(--color-primary)]/40 transition-all disabled:opacity-45 disabled:cursor-not-allowed cursor-pointer shrink-0"
          >
            <Printer size={15} /> {t("history.printDailyReport")}
          </button>
        )}
      </div>

      {/* Summary Strip */}
      {data?.summary && (
        <div className="grid grid-cols-3 gap-3">
          {[
            { label: t("history.totalOrders"), value: data.summary.total_orders },
            { label: t("history.paidOrders"), value: data.summary.paid_orders, primary: true },
            {
              label: t("history.revenue"),
              value: formatUSD(data.summary.total_revenue),
              sub: formatKHR(data.summary.total_revenue),
              primary: true,
            },
          ].map(({ label, value, sub, primary }) => (
            <div
              key={label}
              className="bg-[var(--color-surface)] border border-[var(--color-border)] rounded-2xl p-3.5 text-center shadow-2xs"
            >
              <div
                className={cn(
                  "text-lg sm:text-xl font-bold tabular-nums",
                  primary ? "text-[var(--color-primary)]" : "text-[var(--color-text)]"
                )}
              >
                {value}
              </div>
              {sub && <div className="text-[11px] text-[var(--color-muted)] tabular-nums">{sub}</div>}
              <div className="text-[11px] font-medium text-[var(--color-muted)] mt-0.5">{label}</div>
            </div>
          ))}
        </div>
      )}

      {/* Results */}
      {loading ? (
        <LoadingState text={t("cashier.loadingOrders")} />
      ) : error ? (
        <div className="text-center py-16 text-[var(--color-danger)] text-sm">{error}</div>
      ) : !data || data.orders.length === 0 ? (
        <EmptyState
          icon={Package}
          text={`${t("history.noOrdersOn")} ${displayDate}${
            tableLabel ? ` ${t("history.forTable")} ${tableLabel}` : ""
          }`}
        />
      ) : filteredOrders.length === 0 ? (
        <EmptyState icon={Package} text={`${t("history.noOrdersMatching")} "${search}"`} />
      ) : (
        <div className="space-y-2.5">
          <div className="text-xs text-[var(--color-muted)] px-1">
            {displayDate}
            {tableLabel ? ` · ${t("common.table")} ${tableLabel}` : ""}
            {search ? ` · ${filteredOrders.length} ${t("history.matching")} "${search}"` : ""}
          </div>
          {filteredOrders.map((order) => (
            <OrderRow key={order.order_id} order={order} canPrint={canPrint} />
          ))}
        </div>
      )}
    </div>
  )
}
```

### 5. `frontend/src/pages/admin/Menu.jsx`
Adds category pills, revokes `blob:` URLs cleanly, uses shared `normalizeImageToJpg`, and connects `MenuItemSalesDialog` so admins can inspect per-item monthly sales directly from the menu.

```jsx
import { useState, useEffect, useRef } from "react"
import { Search, Plus, Tag, UtensilsCrossed, Archive, X } from "lucide-react"
import { Button } from "@/components/ui"

function revokeIfBlob(url) {

  }
}

export default function Menu() {
  const { t, lang } = useLang()
  usePageTitle(t("nav.menu"))
  const {
    items,
    archivedItems,
    fetchItems,
    fetchArchivedItems,

  const [activeCat, setActiveCat] = useState("All")
  const [search, setSearch] = useState("")
  const [restoringId, setRestoringId] = useState(null)
  const [addOpen, setAddOpen] = useState(false)
  const [editItem, setEditItem] = useState(null)
  const [salesItem, setSalesItem] = useState(null)
  const [addCatOpen, setAddCatOpen] = useState(false)
  const [newCatName, setNewCatName] = useState("")
  const [newCatType, setNewCatType] = useState("food")
  const [form, setForm] = useState({
    item_name: "",
    item_name_km: "",
    category_id: "",
    price: "",

  const [imagePreview, setImagePreview] = useState(null)
  const [saving, setSaving] = useState(false)
  const [errorMsg, setErrorMsg] = useState("")

  const [deletingItem, setDeletingItem] = useState(false)
  const [deleteItemError, setDeleteItemError] = useState("")

  const previewUrlRef = useRef(null)

  const setPreviewSafe = (nextUrl) => {
    revokeIfBlob(previewUrlRef.current)
    previewUrlRef.current = nextUrl
    setImagePreview(nextUrl)

  useEffect(() => {
    fetchItems().catch(() => {})
    fetchArchivedItems().catch(() => {})
    fetchCategories().catch(() => {})
    return () => revokeIfBlob(previewUrlRef.current)
  }, [fetchItems, fetchArchivedItems, fetchCategories])

  const filtered = sourceItems.filter((i) => {

    const cat = (categories || []).find((c) => c.category_id === i.category_id)
    return matchesCat && menuItemMatchesSearch(i, search, cat?.category_name)

      item_name_km: "",
      category_id: categories[0]?.category_id || "",

    setPreviewSafe(null)

      headers: { "Content-Type": "multipart/form-data" },

    if (
      !form.item_name?.trim() ||
      !form.category_id ||
      form.price === "" ||
      isNaN(parsedPrice) ||
      parsedPrice <= 0

      setAddOpen(false)
    } catch (err) {
      setErrorMsg(err?.response?.data?.message || "Failed to add item.")
    } finally {
      setSaving(false)
    }
  }

  const handleUpdate = async () => {
    const parsedPrice = parseFloat(form.price)
    if (!form.item_name?.trim() || form.price === "" || isNaN(parsedPrice) || parsedPrice <= 0) {

      return

    setErrorMsg("")
    try {

        item_name: form.item_name.trim(),

        await uploadItemImage(editItem.menu_item_id, imageFile)
        await fetchItems()
      }
      setPreviewSafe(null)
      setEditItem(null)
    } catch (err) {
      setErrorMsg(err?.response?.data?.message || "Failed to update item.")
    } finally {
      setSaving(false)
    }

    setErrorMsg("")
    try {
      await toggleItem(editItem.menu_item_id, !editItem.available)

      setErrorMsg(err?.response?.data?.message || "Failed to update status.")

      setNewCatName("")

                  "text-[11px] px-1.5 py-0.5 rounded-full font-bold tabular-nums",

                    ? "bg-white/20 text-white"
                    : "bg-[var(--color-surface-subtle)] text-[var(--color-text)]"
                )}
              >
                {count}
              </span>
            </button>
          ))}
        </div>
      </div>

      {/* Scrollable Category Pills */}
      <div className="flex items-center gap-2 overflow-x-auto no-scrollbar pb-1">
        <button
          type="button"
          onClick={() => setActiveCat("All")}
          className={cn(

                active
                  ? "bg-[var(--color-primary)] text-white border-[var(--color-primary)] shadow-2xs"
                  : "bg-[var(--color-surface)] text-[var(--color-text-secondary)] border-[var(--color-border)] hover:border-[var(--color-primary)]/40"

          <Archive size={14} className="text-[var(--color-muted)] shrink-0" />
          <span>{t("menu.archivedBanner")}</span>
        </div>
      )}

      <MenuGrid
        items={filtered}
        isLoading={activeTab === "archived" ? archivedLoading : isLoading}
        categoryName={categoryName}
        onEdit={openEdit}
        onDelete={requestDeleteItem}
        onViewSales={setSalesItem}
        archivedMode={activeTab === "archived"}
        onRestore={handleRestoreItem}
        restoringId={restoringId}
      />

      <MenuFormDialog
```

```jsx
          setCustomFrom={setCustomFrom}

        />
      )}

  )
}
```

```jsx
export function getDateRangeForPeriod(period, customFrom, customTo) {
  const now = new Date()
  const todayStr = toLocalDateStr(now)

  if (period === "Daily") {
    return { from: todayStr, to: todayStr, label: `Daily (${todayStr})` }
  }
  if (period === "Yesterday") {
    const yest = new Date(now)
    yest.setDate(now.getDate() - 1)
    const yestStr = toLocalDateStr(yest)
    return { from: yestStr, to: yestStr, label: `Yesterday (${yestStr})` }
  }
  if (period === "Weekly") {
    const start = new Date(now)
    start.setDate(now.getDate() - 6)
    const fromStr = toLocalDateStr(start)
    return { from: fromStr, to: todayStr, label: `Last 7 Days (${fromStr} → ${todayStr})` }
  }
  if (period === "Monthly") {
    const fromStr = toLocalDateStr(new Date(now.getFullYear(), now.getMonth(), 1))
    return { from: fromStr, to: todayStr, label: `This Month (${fromStr} → ${todayStr})` }
  }
  const from = customFrom || todayStr
  const to = customTo || todayStr
  return {
    from: from <= to ? from : to,
    to: from <= to ? to : from,
    label: from === to ? `Custom (${from})` : `Custom (${from} → ${to})`,
  }
}

export function filterOrdersByRange(orders = [], fromStr, toStr) {
  const start = new Date(`${fromStr}T00:00:00`)
  const end = new Date(`${toStr}T23:59:59.999`)
  return orders.filter((o) => {
    const d = new Date(o.order_date)
    return d >= start && d <= end
  })
}

export function filterPaymentsByRange(payments = [], fromStr, toStr) {
  const start = new Date(`${fromStr}T00:00:00`)
  const end = new Date(`${toStr}T23:59:59.999`)
  return payments.filter((p) => {
    if (!p.payment_date) return false
    const d = new Date(p.payment_date)
    return d >= start && d <= end
  })
}

                key={p}
                type="button"
                onClick={() => setActivePeriod(p)}
                className={cn(
                  "flex items-center gap-1.5 px-3.5 py-2 rounded-xl border text-xs font-semibold transition-all cursor-pointer",
                  activePeriod === p
                    ? "bg-[var(--color-primary)] text-white border-[var(--color-primary)] shadow-2xs"
                    : "border-[var(--color-border)] bg-[var(--color-surface)] text-[var(--color-text-secondary)] hover:border-[var(--color-primary)]/40 hover:text-[var(--color-primary)]"
                )}
              >
                {p === "Custom" ? <Calendar size={13} /> : <TrendingUp size={13} />}
                {periodLabelKey[p] ? t(periodLabelKey[p]) : p}
              </button>
            ))}
          </div>
        </div>

        {activePeriod === "Custom" && (
          <div className="mt-4 pt-4 border-t border-[var(--color-border)] flex flex-wrap items-center gap-3">
            <div className="flex items-center gap-2 bg-[var(--color-background)] border border-[var(--color-border)] rounded-xl px-3 py-2">
              <Calendar size={14} className="text-[var(--color-primary)] shrink-0" />
              <span className="text-xs font-medium text-[var(--color-muted)]">{t("reports.from")}</span>
              <input
                type="date"
                value={customFrom}
                max={customTo || todayStr}
                onChange={(e) => e.target.value && setCustomFrom(e.target.value)}
                className="text-xs font-semibold text-[var(--color-text)] bg-transparent outline-none cursor-pointer"
              />
            </div>

            <ArrowRight size={14} className="text-[var(--color-muted)] hidden sm:block" />

            <div className="flex items-center gap-2 bg-[var(--color-background)] border border-[var(--color-border)] rounded-xl px-3 py-2">
              <Calendar size={14} className="text-[var(--color-primary)] shrink-0" />
              <span className="text-xs font-medium text-[var(--color-muted)]">{t("reports.to")}</span>
              <input
                type="date"
                value={customTo}
                min={customFrom}
                max={todayStr}
                onChange={(e) => e.target.value && setCustomTo(e.target.value)}
                className="text-xs font-semibold text-[var(--color-text)] bg-transparent outline-none cursor-pointer"
              />
            </div>

            <button
              type="button"
              onClick={() => {
                setCustomFrom(todayStr)
                setCustomTo(todayStr)
              }}
              className="text-xs font-semibold text-[var(--color-primary)] hover:underline px-2 py-1 cursor-pointer"
            >
              {t("reports.resetToday")}
            </button>
          </div>
        )}
      </Card>

      <Card>
        <div className="p-5 pb-4 border-b border-[var(--color-border)] flex flex-col gap-3">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
            <div>
              <div className="font-semibold text-[var(--color-text)]">
                {t("payments.history")} ({filtered.length})
              </div>
              <div className="text-xs text-[var(--color-muted)] mt-0.5 tabular-nums">
                {t("common.total")}:{" "}
                <span className="font-bold text-[var(--color-primary)]">
                  {formatUSD(filteredTotal)}
                </span>{" "}
                ({formatKHR(filteredTotal)})
              </div>
            </div>

            <div className="flex items-center gap-2 flex-wrap">
              <div className="relative w-full sm:w-64">
                <Search
                  size={14}
                  className="absolute left-3 top-1/2 -translate-y-1/2 text-[var(--color-muted)]"
                />
                <Input
                  value={search}
                  onChange={(e) => setSearch(e.target.value)}
                  placeholder={t("payments.searchPlaceholder")}
                  className="pl-9 h-9 text-xs"
                />
              </div>
              <Button
                variant="outline"
                size="sm"
                className="gap-1.5 h-9"
                disabled={filtered.length === 0}
                onClick={() => exportPaymentsExcel(filtered, range.label)}
              >
                <FileSpreadsheet size={14} /> {lang === "km" ? "នាំចេញ Excel" : "Export Excel"}
              </Button>
            </div>
          </div>

          <div className="flex items-center gap-1.5 flex-wrap pt-1">
            {methodNames.map((m) => (
              <button
                key={m}
                type="button"
                onClick={() => setMethodFilter(m)}
                className={cn(
                  "px-3 py-1 rounded-lg text-xs font-semibold border transition-colors cursor-pointer",
                  methodFilter === m
                    ? "bg-[var(--color-primary)] text-white border-[var(--color-primary)]"
                    : "border-[var(--color-border)] text-[var(--color-text-secondary)] hover:bg-[var(--color-surface-subtle)]"
                )}
              >
                {m === "All" ? t("common.all") : m}
              </button>
            ))}
          </div>
        </div>

        <div className="p-2">
          {isLoading ? (
            <div className="text-center text-sm text-[var(--color-muted)] py-10">
              {t("common.loading")}
            </div>
          ) : (
            <Table>
              <TableHeader>
                <TableRow className="hover:bg-transparent border-none">
                  <TableHead>#</TableHead>
                  <TableHead>{t("report.orderNum")}</TableHead>
                  <TableHead>{t("common.table")}</TableHead>
                  <TableHead>{t("pay.method")}</TableHead>
                  <TableHead>{t("payments.cashier")}</TableHead>
                  <TableHead>{t("payments.dateTime")}</TableHead>
                  <TableHead>{t("common.status")}</TableHead>
                  <TableHead className="text-right">{t("payments.amount")}</TableHead>
                  <TableHead className="text-right">{t("common.actions")}</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {filtered.map((p) => {
                  const amt = Number(p.amount || 0)
                  const ordId = p.order_id || p.order?.order_id
                  const linkedOrder = resolveOrder(p)
                  return (
                    <TableRow key={p.payment_id}>
                      <TableCell className="text-[var(--color-muted)] text-xs tabular-nums">
                        PAY-{String(p.payment_id).padStart(4, "0")}
                      </TableCell>
                      <TableCell className="font-semibold tabular-nums">
                        {ordId ? `ORD-${String(ordId).padStart(4, "0")}` : "—"}
                      </TableCell>
                      <TableCell className="text-[var(--color-muted)]">
                        {linkedOrder?.table?.table_number || p.order?.table?.table_number || "—"}
                      </TableCell>
                      <TableCell>
                        <div className="text-xs font-medium text-[var(--color-text-secondary)]">
                          {p.method?.method_name || "—"}
                        </div>
                        {p.external_ref && (
                          <div
                            className="text-[10px] font-mono text-[var(--color-muted)]"
                            title={`Bakong MD5: ${p.external_ref}`}
                          >
                            MD5: {p.external_ref.slice(0, 10)}...
                          </div>
                        )}
                      </TableCell>
                      <TableCell className="text-xs text-[var(--color-muted)]">
                        {p.user?.username || "—"}
                      </TableCell>
                      <TableCell className="text-xs text-[var(--color-muted)]">
                        {p.payment_date ? new Date(p.payment_date).toLocaleString() : "—"}
                      </TableCell>
                      <TableCell>
                        <Badge
                          variant={
                            p.status === "Completed"
                              ? "success"
                              : p.status === "Failed"
                                ? "danger"
                                : "warning"
                          }
                        >
                          {p.status}
                        </Badge>
                      </TableCell>
                      <TableCell className="text-right tabular-nums">
                        <div className="font-bold text-[var(--color-primary)]">{formatUSD(amt)}</div>
                        <div className="text-[10px] text-[var(--color-muted)]">{formatKHR(amt)}</div>
                      </TableCell>
                      <TableCell className="text-right">
                        <div className="flex items-center justify-end gap-1">
                          {linkedOrder?.order_items && (
                            <>
                              <button
                                type="button"
                                onClick={() => printOrderReceipt(linkedOrder)}
                                title={t("cashier.printReceipt")}
                                aria-label={t("cashier.printReceipt")}
                                className="p-1.5 rounded-lg border border-[var(--color-border)] text-[var(--color-muted)] hover:text-[var(--color-primary)] hover:bg-[var(--color-primary-muted)] transition-colors cursor-pointer"
                              >
                                <Printer size={14} />
                              </button>
                              <button
                                type="button"
                                onClick={() => setViewOrder(linkedOrder)}
                                title={t("cashier.viewDetails")}
                                aria-label={t("cashier.viewDetails")}
                                className="p-1.5 rounded-lg border border-[var(--color-border)] text-[var(--color-muted)] hover:text-[var(--color-primary)] hover:bg-[var(--color-primary-muted)] transition-colors cursor-pointer"
                              >
                                <Eye size={14} />
                              </button>
                            </>
                          )}
                        </div>
                      </TableCell>
                    </TableRow>
                  )
                })}
                {filtered.length === 0 && (
                  <TableRow>
                    <TableCell colSpan={9} className="text-center text-sm text-[var(--color-muted)] py-10">
                      {t("payments.noPayments")}
                    </TableCell>
                  </TableRow>
                )}
              </TableBody>
            </Table>
          )}
        </div>

        <ViewOrderDialog order={viewOrder} onClose={() => setViewOrder(null)} />
      </Card>
    </div>
  )
}

export function ReportBuilder({
  activePeriod = "Daily",
  setActivePeriod,
  customFrom,
  setCustomFrom,
  customTo,
  setCustomTo,
  orders = [],
  ordersLoading = false,
  periods: periodsProp,
  printFormat = "a4",
}) {
  const { t, lang } = useLang()
  const periodOptions = periodsProp || periods
  const periodLabelKey = {
    Daily: "reports.daily",
    Yesterday: "reports.yesterday",
    Weekly: "reports.weekly",
    Monthly: "reports.monthly",
    Custom: "reports.custom",
  }
  const todayStr = toLocalDateStr()

  const [localPeriod, setLocalPeriod] = useState(activePeriod)
  const [localFrom, setLocalFrom] = useState(todayStr)
  const [localTo, setLocalTo] = useState(todayStr)
  const [statusFilter, setStatusFilter] = useState("All")
  const [search, setSearch] = useState("")
  const [viewOrder, setViewOrder] = useState(null)

  const currentPeriod = setActivePeriod ? activePeriod : localPeriod
  const handlePeriodChange = setActivePeriod || setLocalPeriod
  const fromDate = customFrom !== undefined ? customFrom : localFrom
  const toDate = customTo !== undefined ? customTo : localTo
  const handleFromChange = setCustomFrom || setLocalFrom
  const handleToChange = setCustomTo || setLocalTo

  const range = useMemo(
    () => getDateRangeForPeriod(currentPeriod, fromDate, toDate),
    [currentPeriod, fromDate, toDate]
  )

  const rangedOrders = useMemo(
    () => filterOrdersByRange(orders, range.from, range.to),
    [orders, range.from, range.to]
  )
  const paidOrders = useMemo(() => rangedOrders.filter((o) => o.status === "Paid"), [rangedOrders])
  const totalRevenue = useMemo(() => paidOrders.reduce((s, o) => s + orderTotal(o), 0), [paidOrders])
  const avgOrderValue = paidOrders.length ? totalRevenue / paidOrders.length : 0
  const itemsSold = useMemo(
    () =>
      paidOrders.reduce(
        (sum, o) => sum + (o.order_items?.reduce((s, i) => s + i.quantity, 0) || 0),
        0
      ),
    [paidOrders]
  )

  const visibleOrders = useMemo(() => {
    const q = search.trim().toLowerCase()
    return rangedOrders.filter((o) => {
      if (statusFilter === "Paid" && o.status !== "Paid") return false
      if (statusFilter === "Unpaid" && o.status === "Paid") return false
      if (!q) return true
      const idStr = String(o.order_id)
      const padded = idStr.padStart(4, "0")
      const tbl = String(o.table?.table_number || "").toLowerCase()
      return idStr.includes(q) || padded.includes(q) || `ord-${padded}`.includes(q) || tbl.includes(q)
    })
  }, [rangedOrders, statusFilter, search])

  const kpis = [
    {
      label: t("reports.periodRevenue"),
      value: formatUSD(totalRevenue),
      sub: `${formatKHR(totalRevenue)} KHR`,
      icon: DollarSign,
      featured: true,
    },
    {
      label: t("reports.paidOrders"),
      value: paidOrders.length,
      sub: `${paidOrders.length} / ${rangedOrders.length} ${t("cashier.orders")}`,
      icon: CheckCircle2,
    },
    {
      label: t("reports.avgOrderValue"),
      value: formatUSD(avgOrderValue),
      sub: formatKHR(avgOrderValue),
      icon: TrendingUp,
    },
    {
      label: t("reports.itemsSold"),
      value: itemsSold,
      sub: `${paidOrders.length} ${t("reports.paidOrders")}`,
      icon: ShoppingCart,
    },
  ]

  return (
    <div className="space-y-5">
      <Card className="p-5">
        <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-4">
          <div className="flex items-center gap-2.5">
            <div className="w-8 h-8 rounded-xl bg-[var(--color-primary-muted)] flex items-center justify-center">
              <BarChart3 size={16} className="text-[var(--color-primary)]" />
            </div>
            <div>
              <div className="font-semibold text-[var(--color-text)] text-base">
                {t("report.salesReport")}
              </div>
              <div className="text-xs text-[var(--color-muted)]">{range.label}</div>
            </div>
          </div>

          {periodOptions.length > 1 && (
            <div className="flex flex-wrap items-center gap-2">
              {periodOptions.map((p) => (
                <button
                  key={p}
                  type="button"
                  onClick={() => handlePeriodChange(p)}
                  className={cn(
                    "flex items-center gap-1.5 px-3.5 py-2 rounded-xl border text-xs font-semibold transition-all cursor-pointer",
                    currentPeriod === p
                      ? "bg-[var(--color-primary)] text-white border-[var(--color-primary)] shadow-2xs"
                      : "border-[var(--color-border)] bg-[var(--color-surface)] text-[var(--color-text-secondary)] hover:border-[var(--color-primary)]/40 hover:text-[var(--color-primary)]"
                  )}
                >
                  {p === "Custom" ? <Calendar size={13} /> : <TrendingUp size={13} />}
                  {periodLabelKey[p] ? t(periodLabelKey[p]) : p}
                </button>
              ))}
            </div>
          )}
        </div>

        {currentPeriod === "Custom" && (
          <div className="mt-4 pt-4 border-t border-[var(--color-border)] flex flex-wrap items-center gap-3">
            <div className="flex items-center gap-2 bg-[var(--color-background)] border border-[var(--color-border)] rounded-xl px-3 py-2">
              <Calendar size={14} className="text-[var(--color-primary)] shrink-0" />
              <span className="text-xs font-medium text-[var(--color-muted)]">{t("reports.from")}</span>
              <input
                type="date"
                value={fromDate}
                max={toDate || todayStr}
                onChange={(e) => e.target.value && handleFromChange(e.target.value)}
                className="text-xs font-semibold text-[var(--color-text)] bg-transparent outline-none cursor-pointer"
              />
            </div>

            <ArrowRight size={14} className="text-[var(--color-muted)] hidden sm:block" />

            <div className="flex items-center gap-2 bg-[var(--color-background)] border border-[var(--color-border)] rounded-xl px-3 py-2">
              <Calendar size={14} className="text-[var(--color-primary)] shrink-0" />
              <span className="text-xs font-medium text-[var(--color-muted)]">{t("reports.to")}</span>
              <input
                type="date"
                value={toDate}
                min={fromDate}
                max={todayStr}
                onChange={(e) => e.target.value && handleToChange(e.target.value)}
                className="text-xs font-semibold text-[var(--color-text)] bg-transparent outline-none cursor-pointer"
              />
            </div>

            <button
              type="button"
              onClick={() => {
                handleFromChange(todayStr)
                handleToChange(todayStr)
              }}
              className="text-xs font-semibold text-[var(--color-primary)] hover:underline px-2 py-1 cursor-pointer"
            >
              {t("reports.resetToday")}
            </button>
          </div>
        )}
      </Card>

      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
        {kpis.map(({ label, value, sub, icon: Icon, featured }) => (
          <Card
            key={label}
            className={cn(
              "p-5",
              featured && "border-[var(--color-primary)]/35 bg-[var(--color-primary-muted)]/35"
            )}
          >
            <div className="flex items-center justify-between mb-2">
              <span className="text-[11px] font-semibold text-[var(--color-muted)] uppercase tracking-wider">
                {label}
              </span>
              <div
                className={cn(
                  "w-8 h-8 rounded-xl flex items-center justify-center",
                  featured
                    ? "bg-[var(--color-primary)] text-white"
                    : "bg-[var(--color-surface-subtle)] text-[var(--color-primary)]"
                )}
              >
                <Icon size={15} />
              </div>
            </div>
            <div
              className={cn(
                "font-display text-2xl font-bold tabular-nums",
                featured ? "text-[var(--color-primary)]" : "text-[var(--color-text)]"
              )}
            >
              {value}
            </div>
            <div className="text-xs text-[var(--color-muted)] mt-0.5">{sub}</div>
          </Card>
        ))}
      </div>

      <Card>
        <div className="p-5 border-b border-[var(--color-border)] flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div>
            <div className="font-semibold text-[var(--color-text)]">
              {t("reports.orderTransactions")} ({visibleOrders.length})
            </div>
            <div className="text-xs text-[var(--color-muted)]">
              {t("reports.detailedBreakdown")} {range.label}
            </div>
          </div>

          <div className="flex flex-wrap items-center gap-2">
            <Button
              variant="outline"
              size="sm"
              className="gap-1.5"
              disabled={visibleOrders.length === 0}
              onClick={() => exportOrdersCsv(visibleOrders, range.label)}
            >
              <Download size={13} /> {t("reports.exportCsv")}
            </Button>
            <Button
              variant="outline"
              size="sm"
              className="gap-1.5"
              disabled={visibleOrders.length === 0}
              onClick={() =>
                exportOrdersExcel({
                  orders: visibleOrders,
                  periodLabel: range.label,
                  totalRevenue,
                  paidOrdersCount: paidOrders.length,
                  avgOrderValue,
                })
              }
            >
              <FileSpreadsheet size={13} /> {lang === "km" ? "នាំចេញ Excel" : "Export Excel"}
            </Button>
            <Button
              size="sm"
              className="gap-1.5"
              disabled={rangedOrders.length === 0}
              onClick={() => {
                const fn = printFormat === "thermal" ? printThermalReport : printExecutiveReport
                fn({
                  period: range.label,
                  totalRevenue,
                  paidOrders,
                  avgOrderValue,
                  periodOrders: rangedOrders,
                })
              }}
            >
              <Printer size={13} /> {t("reports.printReport")}
            </Button>
          </div>
        </div>

        <div className="px-5 py-3 border-b border-[var(--color-border)] flex flex-wrap items-center justify-between gap-3 bg-[var(--color-background)]/45">
          <div className="relative flex-1 min-w-[200px] max-w-xs">
            <Search
              size={14}
              className="absolute left-3 top-1/2 -translate-y-1/2 text-[var(--color-muted)]"
            />
            <Input
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              placeholder={t("reports.searchOrderOrTable")}
              className="pl-9 h-9 text-xs"
            />
          </div>

          <div className="flex items-center gap-1.5">
            {["All", "Paid", "Unpaid"].map((st) => (
              <button
                key={st}
                type="button"
                onClick={() => setStatusFilter(st)}
                className={cn(
                  "px-3 py-1.5 rounded-lg text-xs font-semibold transition-colors cursor-pointer",
                  statusFilter === st
                    ? "bg-[var(--color-primary)] text-white"
                    : "text-[var(--color-text-secondary)] hover:bg-[var(--color-surface-subtle)]"
                )}
              >
                {st === "All"
                  ? t("reports.all")
                  : st === "Paid"
                    ? t("reports.paid")
                    : t("reports.unpaid")}
              </button>
            ))}
          </div>
        </div>

        <div className="p-2">
          {ordersLoading ? (
            <div className="text-center text-sm text-[var(--color-muted)] py-10">
              {t("cashier.loadingOrders")}
            </div>
          ) : (
            <Table>
              <TableHeader>
                <TableRow className="hover:bg-transparent border-none">
                  <TableHead>{t("report.orderNum")}</TableHead>
                  <TableHead>{t("common.table")}</TableHead>
                  <TableHead>{t("payments.dateTime")}</TableHead>
                  <TableHead>{t("common.items")}</TableHead>
                  <TableHead>{t("report.payment")}</TableHead>
                  <TableHead>{t("common.status")}</TableHead>
                  <TableHead className="text-right">{t("common.total")}</TableHead>
                  <TableHead className="text-right">{t("common.actions")}</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {visibleOrders.slice(0, 100).map((o) => {
                  const total = orderTotal(o)
                  const itemCount = o.order_items?.reduce((s, i) => s + i.quantity, 0) || 0
                  const method = o.payments?.[0]?.method?.method_name || "—"
                  return (
                    <TableRow key={o.order_id}>
                      <TableCell className="font-semibold tabular-nums">
                        ORD-{String(o.order_id).padStart(4, "0")}
                      </TableCell>
                      <TableCell className="text-[var(--color-muted)]">
                        {o.table?.table_number || "—"}
                      </TableCell>
                      <TableCell className="text-[var(--color-muted)] text-xs">
                        {new Date(o.order_date).toLocaleString()}
                      </TableCell>
                      <TableCell className="text-[var(--color-muted)]">
                        {itemCount} {t("station.items")}
                      </TableCell>
                      <TableCell className="text-xs text-[var(--color-text-secondary)]">
                        {method}
                      </TableCell>
                      <TableCell>
                        <Badge variant={o.status === "Paid" ? "success" : "warning"}>
                          {o.status}
                        </Badge>
                      </TableCell>
                      <TableCell className="text-right tabular-nums">
                        <div className="font-bold text-[var(--color-primary)]">
                          {formatUSD(total)}
                        </div>
                        <div className="text-[10px] text-[var(--color-muted)]">
                          {formatKHR(total)}
                        </div>
                      </TableCell>
                      <TableCell className="text-right">
                        <div className="flex items-center justify-end gap-1">
                          <button
                            type="button"
                            onClick={() => printOrderReceipt(o)}
                            title={t("cashier.printReceipt")}
                            aria-label={t("cashier.printReceipt")}
                            className="p-1.5 rounded-lg border border-[var(--color-border)] text-[var(--color-muted)] hover:text-[var(--color-primary)] hover:bg-[var(--color-primary-muted)] transition-colors cursor-pointer"
                          >
                            <Printer size={14} />
                          </button>
                          <button
                            type="button"
                            onClick={() => setViewOrder(o)}
                            title={t("cashier.viewDetails")}
                            aria-label={t("cashier.viewDetails")}
                            className="p-1.5 rounded-lg border border-[var(--color-border)] text-[var(--color-muted)] hover:text-[var(--color-primary)] hover:bg-[var(--color-primary-muted)] transition-colors cursor-pointer"
                          >
                            <Eye size={14} />
                          </button>
                        </div>
                      </TableCell>
                    </TableRow>
                  )
                })}
                {visibleOrders.length === 0 && (
                  <TableRow>
                    <TableCell colSpan={8} className="text-center text-sm text-[var(--color-muted)] py-10">
                      {t("reports.noOrdersRange")}
                    </TableCell>
                  </TableRow>
                )}
              </TableBody>
            </Table>
          )}
        </div>
        {visibleOrders.length > 100 && (
          <div className="px-5 pb-4 text-xs text-[var(--color-muted)]">
            {t("reports.showingFirst100")} {visibleOrders.length} {t("reports.exportCsvHint")}
```

### 11. `frontend/src/components/ErrorBoundary.jsx`
Bilingual-aware (reads `localStorage.getItem("zoom_lang")` safely since `ErrorBoundary` wraps `<LanguageProvider>`).

```jsx
import React from "react"
import { AlertTriangle, RefreshCw, Home } from "lucide-react"
import { Button } from "./ui"

export class ErrorBoundary extends React.Component {
  constructor(props) {
    super(props)
    this.state = { hasError: false, error: null }
  }

  static getDerivedStateFromError(error) {
    return { hasError: true, error }
  }

  componentDidCatch(error, errorInfo) {
    console.error("ErrorBoundary caught an unhandled UI error:", error, errorInfo)
  }

  handleReset = () => {
    this.setState({ hasError: false, error: null })
    window.location.reload()
  }

  render() {
    if (this.state.hasError) {
      const isKm =
        typeof window !== "undefined" && localStorage.getItem("zoom_lang") === "km"

      return (
        <div className="min-h-dvh w-full flex items-center justify-center p-6 bg-[var(--color-background)]">
          <div className="max-w-md w-full rounded-3xl border border-[var(--color-border)] bg-[var(--color-surface)] p-8 shadow-xl text-center space-y-5">
            <div className="w-14 h-14 mx-auto rounded-2xl bg-[var(--color-danger-muted)] flex items-center justify-center text-[var(--color-danger)]">
              <AlertTriangle size={26} />
            </div>

            <div className="space-y-1.5">
              <h2 className="text-xl font-bold font-display text-[var(--color-text)]">
                {isKm ? "មានបញ្ហាបច្ចេកទេសបន្តិចបន្តួច" : "Something went wrong"}
              </h2>
              <p className="text-xs text-[var(--color-muted)] leading-relaxed">
                {isKm
                  ? "មានបញ្ហាដែលមិនបានរំពឹងទុកបានកើតឡើង។ អ្នកអាចផ្ទុកទំព័រឡើងវិញ ឬត្រឡប់ទៅទំព័រដើម។"
                  : "An unexpected interface error occurred. You can reload this view or return to the home screen."}
              </p>
            </div>

            {this.state.error?.message && (
              <div className="p-3 rounded-xl bg-[var(--color-danger-muted)]/60 border border-[var(--color-danger)]/20 text-[11px] font-mono text-[var(--color-danger)] text-left overflow-x-auto">
                {this.state.error.message}
              </div>
            )}

            <div className="flex flex-col sm:flex-row gap-2.5 pt-1">
              <Button
                variant="outline"
                className="flex-1"
                onClick={() => {
                  window.location.href = "/"
                }}
              >
                <Home size={15} /> {isKm ? "ទំព័រដើម" : "Return Home"}
              </Button>
              <Button className="flex-1" onClick={this.handleReset}>
                <RefreshCw size={15} /> {isKm ? "ផ្ទុកឡើងវិញ" : "Reload View"}
              </Button>
            </div>
          </div>
        </div>
      )
    }

    return this.props.children
  }
}

export default ErrorBoundary
```

### 12. `frontend/vite.config.js`

```js
import { defineConfig } from "vite"
import react from "@vitejs/plugin-react"
import tailwindcss from "@tailwindcss/vite"
import path from "path"

const devProxyTarget = process.env.VITE_DEV_PROXY_TARGET || "http://localhost:8080"
const allowedHosts = (process.env.VITE_ALLOWED_HOSTS || "")
  .split(",")
  .map((h) => h.trim())
  .filter(Boolean)

export default defineConfig({
  plugins: [react(), tailwindcss()],
  resolve: {
    alias: {
      "@": path.resolve(import.meta.dirname, "./src"),
    },

    port: 3000,
    strictPort: false,
    host: true,
    ...(allowedHosts.length > 0 ? { allowedHosts } : {}),
    proxy: {
      "/api": {
        target: devProxyTarget,
        changeOrigin: true,
      },
      "/uploads": {
        target: devProxyTarget,
        changeOrigin: true,
      },
      "/socket.io": {
        target: devProxyTarget,
        ws: true,
        changeOrigin: true,
      },
    },
  },
})
```
