Here is **Phase 4: Kitchen & Barista KDS Ticket Board, Station Menu & Stock, and De-duplicated Live Notifications**.

### What's improved in Phase 4
- **De-duplicated & Debounced Station Socket Fetching (`useStaffNotifications.js`):**
  - Collapses multi-item order bursts into **1 single 250ms-debounced queue fetch** (eliminating the ~12 duplicate fetches per order), also listens for `order_item:updated` and `order_item:note_changed` (cashier edits), and cleans up bell timers on unmount.
- **Real Multi-Column KDS Board (`StationOrders.jsx`):**
  - Open **2–3 column ticket grid** (tickets start **expanded** by default so chefs/baristas immediately see every dish without tapping, plus an Expand/Compact toggle).
  - **Live elapsed-time wait timers** (`3m`, `12m`, `24m`) that automatically shift to **amber at 10m** and **red at 20m**.
  - **Double-tap lock** on both whole-ticket and per-item action buttons so busy touchscreens never skip statuses or fire duplicate PATCH calls.
  - **Per-item status pills + individual item bump buttons** (great when a table adds a 2nd round of `Pending` items to an already `Preparing` ticket).
  - Unmistakable full-width red **Allergy Warning Banner** (supports both English & Khmer `អាលែកហ្ស៊ី`).
- **Upgraded Station Menu (`StationMenu.jsx`):**
  - One-tap **horizontal category pills** + **All / Available / Sold Out** filter tabs with live counts, cleaner toggle controls, and USD/KHR display.
- **Upgraded Station Stock (`StationStock.jsx`):**
  - Search bar + **Low Stock Only** filter, compact card layout, quick quantity steppers (`+0.5`, `+1`, `+5`), live **remaining stock preview**, and full English/Khmer copy.
- **Accessible Shared Modal (`shared/Modal.jsx`):** Adds `Escape` key closing, scroll lock, and `aria-label`.

```js
import { useState, useCallback, useEffect, useRef } from "react"
import { useSocketRole, useSocketEvent } from "../services/socket"
import { useOrderItemStore } from "../context/orderItemContext"

let sharedAudioCtx = null
let lastSoundPlayedAt = 0
const SOUND_COOLDOWN_MS = 3500
const ORDER_BATCH_WINDOW_MS = 8000
const recentOrderRingMap = new Map()

function shouldRingForOrder(orderId, eventGroup = "new") {
  const now = Date.now()
  for (const [k, ts] of recentOrderRingMap.entries()) {
    if (now - ts > ORDER_BATCH_WINDOW_MS) {
      recentOrderRingMap.delete(k)
    }
  }
  if (!orderId) return true
  const key = `${eventGroup}:${orderId}`
  const prev = recentOrderRingMap.get(key)
  recentOrderRingMap.set(key, now)
  if (prev && now - prev < ORDER_BATCH_WINDOW_MS) {
    return false
  }
  return true
}

function getAudioContext() {
  if (!sharedAudioCtx) {
    const Ctx = window.AudioContext || window.webkitAudioContext
    if (!Ctx) return null
    sharedAudioCtx = new Ctx()
  }
  return sharedAudioCtx
}

export function isAudioUnlocked() {
  return !!sharedAudioCtx && sharedAudioCtx.state === "running"
}

export function unlockAudioNow() {
  try {
    const ctx = getAudioContext()
    if (!ctx) return
    if (ctx.state === "suspended") {
      ctx.resume().then(() => playAlertSound({ force: true })).catch(() => {})
    } else {
      playAlertSound({ force: true })
    }
  } catch (err) {
    console.warn("Could not unlock audio:", err)
  }
}

export function playAlertSound({ force = false } = {}) {
  try {
    const nowMs = Date.now()
    if (!force && nowMs - lastSoundPlayedAt < SOUND_COOLDOWN_MS) {
      return
    }
    lastSoundPlayedAt = nowMs

    const ctx = getAudioContext()
    if (!ctx) return

    const scheduleChime = () => {
      const playTone = (freq, startTime, duration, volume = 0.5) => {
        try {
          const osc = ctx.createOscillator()
          const gain = ctx.createGain()
          osc.connect(gain)
          gain.connect(ctx.destination)
          osc.frequency.value = freq
          osc.type = "sine"
          const t0 = Math.max(startTime, ctx.currentTime + 0.01)
          gain.gain.setValueAtTime(volume, t0)
          gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration)
          osc.start(t0)
          osc.stop(t0 + duration)
        } catch {}
      }
      const now = ctx.currentTime + 0.02
      playTone(880, now, 0.22)
      playTone(1175, now + 0.24, 0.32)
    }

    if (ctx.state === "suspended") {
      ctx.resume().then(scheduleChime).catch(() => {})
    } else {
      scheduleChime()
    }
  } catch (err) {
    console.warn("Could not play alert sound:", err)
  }
}

export function unlockAudioOnFirstInteraction() {
  const unlock = () => {
    try {
      const ctx = getAudioContext()
      if (ctx) {
        if (ctx.state === "suspended") ctx.resume()
        const buffer = ctx.createBuffer(1, 1, 22050)
        const source = ctx.createBufferSource()
        source.buffer = buffer
        source.connect(ctx.destination)
        source.start(0)
      }
    } catch {}
    window.removeEventListener("click", unlock)
    window.removeEventListener("keydown", unlock)
    window.removeEventListener("touchstart", unlock)
  }
  window.addEventListener("click", unlock, { once: true })
  window.addEventListener("keydown", unlock, { once: true })
  window.addEventListener("touchstart", unlock, { once: true })
}

export function useStaffNotifications(role) {
  const { connected } = useSocketRole(role)
  const [bellRinging, setBellRinging] = useState(false)
  const bellTimerRef = useRef(null)
  const refetchTimerRef = useRef(null)

  const { kitchenQueue, baristaQueue, fetchKitchenQueue, fetchBaristaQueue } =
    useOrderItemStore()

  // Single debounced queue refresh shared by Layout bell + StationOrders
  const scheduleQueueFetch = useCallback(() => {
    if (role !== "kitchen" && role !== "barista") return
    if (refetchTimerRef.current) clearTimeout(refetchTimerRef.current)
    refetchTimerRef.current = setTimeout(() => {
      refetchTimerRef.current = null
      if (role === "kitchen") fetchKitchenQueue().catch(() => {})
      if (role === "barista") fetchBaristaQueue().catch(() => {})
    }, 250)
  }, [role, fetchKitchenQueue, fetchBaristaQueue])

  useEffect(() => {
    if (role === "kitchen") fetchKitchenQueue().catch(() => {})
    if (role === "barista") fetchBaristaQueue().catch(() => {})
    return () => {
      if (refetchTimerRef.current) clearTimeout(refetchTimerRef.current)
      if (bellTimerRef.current) clearTimeout(bellTimerRef.current)
    }
  }, [role, fetchKitchenQueue, fetchBaristaQueue])

  const ring = useCallback(() => {
    setBellRinging(true)
    playAlertSound()
    if (bellTimerRef.current) clearTimeout(bellTimerRef.current)
    bellTimerRef.current = setTimeout(() => {
      bellTimerRef.current = null
      setBellRinging(false)
    }, 2000)
  }, [])

  const ringForOrder = useCallback(
    (orderId, eventGroup = "new") => {
      if (!shouldRingForOrder(orderId, eventGroup)) return
      ring()
    },
    [ring]
  )

  useSocketEvent("order:new", (order) => {
    const orderId = order?.order_id
    if (role === "cashier") {
      ringForOrder(orderId, "new")
    }
    if (role === "kitchen" || role === "barista") {
      scheduleQueueFetch()
      ringForOrder(orderId, "new")
    }
  })

  useSocketEvent("order:confirmed", (order) => {
    const orderId = order?.order_id
    if (role === "kitchen" || role === "barista") {
      scheduleQueueFetch()
      ringForOrder(orderId, "new")
    }
  })

  useSocketEvent("order:deleted", () => {
    scheduleQueueFetch()
  })

  useSocketEvent("order_item:new", (item) => {
    const orderId = item?.order_id ?? item?.order?.order_id
    if (role === "cashier") {
      ringForOrder(orderId, "new")
    }
    if (role === "kitchen" || role === "barista") {
      scheduleQueueFetch()
      ringForOrder(orderId, "new")
    }
  })

  useSocketEvent("order_item:updated", () => {
    scheduleQueueFetch()
  })

  useSocketEvent("order_item:note_changed", () => {
    scheduleQueueFetch()
  })

  useSocketEvent("order_item:deleted", () => {
    scheduleQueueFetch()
  })

  useSocketEvent("order_item:status_changed", (item) => {
    scheduleQueueFetch()
    if (item?.status === "Ready" && role === "cashier") {
      const orderId = item?.order_id ?? item?.order?.order_id
      ringForOrder(orderId, "ready")
    }
  })

  useSocketEvent("order:status_changed", (order) => {
    if (order?.status === "Served" && role === "cashier") {
      ringForOrder(order?.order_id, "ready")
    }
  })

  let count = 0
  if (role === "kitchen") count = kitchenQueue.filter((i) => i.status === "Pending").length
  if (role === "barista") count = baristaQueue.filter((i) => i.status === "Pending").length

  const refresh = useCallback(() => {
    if (role === "kitchen") fetchKitchenQueue().catch(() => {})
    if (role === "barista") fetchBaristaQueue().catch(() => {})
  }, [role, fetchKitchenQueue, fetchBaristaQueue])

  return { count, bellRinging, connected, refresh }
}
```

```jsx
function groupByOrder(items) {
  const map = new Map()
  for (const oi of items) {
    const orderId = oi.order?.order_id
    if (!map.has(orderId)) {
      map.set(orderId, {
        order_id: orderId,
        queue_number: oi.order?.queue_number,
        party_size: oi.order?.party_size,
        table_number: oi.order?.table?.table_number || "—",
        order_date: oi.order?.order_date,
        note: oi.order?.note,
        items: [],
      })
    }
    map.get(orderId).items.push(oi)
  }
  return Array.from(map.values()).sort(
    (a, b) => new Date(a.order_date) - new Date(b.order_date)
  )
}

function getWaitMinutes(dateStr, nowMs) {
  if (!dateStr) return 0
  return Math.max(0, Math.floor((nowMs - new Date(dateStr).getTime()) / 60000))
}

function formatWaitLabel(mins) {
  if (mins < 1) return "<1m"
  if (mins < 60) return `${mins}m`
  const h = Math.floor(mins / 60)
  const m = mins % 60
  return m > 0 ? `${h}h ${m}m` : `${h}h`
}
```

```jsx
  const { t, lang } = useLang()
  usePageTitle("Stock")
  const { stockItems, fetchStock, decreaseStock, isLoading } = useStockStore()

  const [search, setSearch] = useState("")
  const [lowOnly, setLowOnly] = useState(false)
  const [expandedId, setExpandedId] = useState(null)
  const [qty, setQty] = useState({})
  const [busy, setBusy] = useState(null)
  const [error, setError] = useState({})
  const [success, setSuccess] = useState({})

  useEffect(() => {
    fetchStock().catch(() => {})
  }, [fetchStock])

  const isLow = (item) =>
    Number(item.quantity) <= 0 ||
    (item.min_level != null && Number(item.quantity) <= Number(item.min_level))

  const lowCount = useMemo(
    () => (stockItems || []).filter(isLow).length,
    [stockItems]
  )

  const filtered = useMemo(() => {
    const q = search.trim().toLowerCase()
    return (stockItems || []).filter((item) => {
      if (lowOnly && !isLow(item)) return false
      if (!q) return true
      return (
        String(item.item_name || "").toLowerCase().includes(q) ||
        String(item.unit?.unit_name || "").toLowerCase().includes(q)
      )
    })
  }, [stockItems, search, lowOnly])

  const adjustQty = (id, delta) => {
    setError((prev) => ({ ...prev, [id]: null }))
    setQty((prev) => {
      const cur = parseFloat(prev[id]) || 0
      const next = Math.max(0, Math.round((cur + delta) * 100) / 100)
      return { ...prev, [id]: next === 0 ? "" : String(next) }
    })
  }

  const handleDecrease = async (item) => {
    const amount = parseFloat(qty[item.stock_id]) || 0
    if (amount <= 0) {
      setError((prev) => ({ ...prev, [item.stock_id]: t("station.enterValidQty") }))
      return
    }
    if (amount > Number(item.quantity)) {
      setError((prev) => ({ ...prev, [item.stock_id]: t("station.notEnoughStock") }))
      return
    }

    setBusy(item.stock_id)
    setError((prev) => ({ ...prev, [item.stock_id]: null }))
    try {
      await decreaseStock(item.stock_id, amount)
      setQty((prev) => ({ ...prev, [item.stock_id]: "" }))
      setSuccess((prev) => ({ ...prev, [item.stock_id]: true }))
      setTimeout(() => {
        setSuccess((prev) => ({ ...prev, [item.stock_id]: false }))
      }, 2200)
      setExpandedId(null)
    } catch (err) {
      setError((prev) => ({
        ...prev,
        [item.stock_id]: err?.response?.data?.message || t("common.failed"),
      }))
    } finally {
      setBusy(null)
    }
  }

  return (
    <div className="space-y-4">
      {/* Header + Search + Low Stock Filter */}
      <div className="bg-[var(--color-surface)] border border-[var(--color-border)] rounded-2xl p-4 shadow-2xs space-y-3">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-[var(--color-primary-muted)] text-[var(--color-primary)] flex items-center justify-center shrink-0">
              <Package size={19} />
            </div>
            <div>
              <h1 className="font-display text-xl font-bold text-[var(--color-text)] leading-tight">
                {t("nav.stock")}
              </h1>
              <p className="text-xs text-[var(--color-muted)] mt-0.5">
                {t("station.stockSubtitle")}
              </p>
            </div>
          </div>

          <button
            type="button"
            onClick={() => setLowOnly((v) => !v)}
            className={cn(
              "inline-flex items-center gap-1.5 px-3.5 py-2 rounded-xl text-xs font-semibold border transition-all cursor-pointer self-start sm:self-auto",
              lowOnly
                ? "bg-[var(--color-danger)] text-white border-[var(--color-danger)]"
                : "bg-[var(--color-background)]/60 border-[var(--color-border)] text-[var(--color-text-secondary)] hover:border-[var(--color-warning)]"
            )}
          >
            <AlertTriangle size={13} />
            <span>
              {t("stock.lowStockOnly")} ({lowCount})
            </span>
          </button>
        </div>

        <div className="relative max-w-md">
          <Search
            size={15}
            className="absolute left-3.5 top-1/2 -translate-y-1/2 text-[var(--color-muted)] pointer-events-none"
          />
          <input
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            placeholder={t("stock.searchPlaceholder")}
            className="w-full h-10 pl-10 pr-9 rounded-xl border border-[var(--color-border)] bg-[var(--color-background)]/60 text-sm text-[var(--color-text)] placeholder:text-[var(--color-muted)]/75 outline-none focus:bg-[var(--color-surface)] focus:ring-2 focus:ring-[var(--color-primary)]/25 focus:border-[var(--color-primary)]"
          />

                )}

                <button
                  type="button"
                  onClick={() => setExpandedId(expanded ? null : item.stock_id)}
                  className="w-full flex items-center gap-3.5 px-4 py-3.5 hover:bg-[var(--color-surface-subtle)]/55 transition-colors text-left cursor-pointer"

                      )}
                    </div>

                    <div className="text-xs text-[var(--color-muted)] mt-1 tabular-nums">
                      <span className="font-bold text-[var(--color-text-secondary)]">
                        {Number(item.quantity).toFixed(2)} {item.unit?.unit_name}
                      </span>
                      {item.min_level != null && (
                        <span>
                          {" "}
                          · {t("stock.minLevel")}: {Number(item.min_level).toFixed(2)}
                        </span>
                      )}
                    </div>
                  </div>

                  {expanded ? (
                    <ChevronUp size={16} className="text-[var(--color-muted)] shrink-0" />
                  ) : (
                    <ChevronDown size={16} className="text-[var(--color-muted)] shrink-0" />
                  )}
                </button>

                {expanded && (
                  <div className="border-t border-[var(--color-border)] bg-[var(--color-background)]/45 px-4 py-3.5 space-y-3">
                    <div className="flex items-center justify-between text-xs">
                      <span className="text-[var(--color-muted)]">
                        {lang === "km"
                          ? `តើអ្នកដកប៉ុន្មាន ${item.unit?.unit_name || ""} ចេញពីឃ្លាំង?`
                          : `How many ${item.unit?.unit_name || "units"} are you taking?`}
                      </span>
                      {takeVal > 0 && (
                        <span
                          className={cn(
                            "font-semibold tabular-nums",
                            remaining < 0
                              ? "text-[var(--color-danger)]"
                              : "text-[var(--color-primary)]"
                          )}
                        >
                          {lang === "km" ? "នៅសល់" : "Left"}: {remaining.toFixed(2)}{" "}
                          {item.unit?.unit_name}
                        </span>
                      )}
                    </div>

                    <div className="flex items-center gap-2">
                      <button
                        type="button"
                        onClick={() => adjustQty(item.stock_id, -1)}
                        aria-label="Decrease quantity"
                        className="w-9 h-9 rounded-xl bg-[var(--color-surface)] border border-[var(--color-border)] flex items-center justify-center text-[var(--color-text-secondary)] hover:border-[var(--color-primary)] cursor-pointer"
                      >
                        <Minus size={14} />
                      </button>

                      <input
                        type="number"
                        min="0"
                        step="0.1"
                        inputMode="decimal"
                        value={qty[item.stock_id] || ""}
                        onChange={(e) => {
                          setError((prev) => ({ ...prev, [item.stock_id]: null }))
                          setQty((prev) => ({ ...prev, [item.stock_id]: e.target.value }))
                        }}
                        placeholder="0"
                        className="flex-1 h-9 text-center text-sm font-bold tabular-nums bg-[var(--color-surface)] border border-[var(--color-border)] rounded-xl px-3 outline-none focus:ring-2 focus:ring-[var(--color-primary)]/25 focus:border-[var(--color-primary)]"
                      />

                      <button
                        type="button"
                        onClick={() => adjustQty(item.stock_id, 1)}
                        aria-label="Increase quantity"
                        className="w-9 h-9 rounded-xl bg-[var(--color-surface)] border border-[var(--color-border)] flex items-center justify-center text-[var(--color-text-secondary)] hover:border-[var(--color-primary)] cursor-pointer"
                      >
                        <Plus size={14} />
                      </button>

                      <span className="text-xs font-semibold text-[var(--color-muted)] min-w-[40px]">
                        {item.unit?.unit_name}
                      </span>
                    </div>

                    {/* Quick Increment Chips */}
                    <div className="flex items-center gap-1.5 flex-wrap">
                      {[0.5, 1, 2, 5].map((step) => (
                        <button
                          key={step}
                          type="button"
                          onClick={() => adjustQty(item.stock_id, step)}
                          className="px-2.5 py-1 rounded-lg bg-[var(--color-surface)] border border-[var(--color-border)] text-xs font-semibold text-[var(--color-text-secondary)] hover:border-[var(--color-primary)] hover:text-[var(--color-primary)] transition-colors cursor-pointer tabular-nums"
                        >
                          +{step}
                        </button>
                      ))}
                    </div>

                    {error[item.stock_id] && (
                      <p role="alert" className="text-xs font-medium text-[var(--color-danger)]">
                        {error[item.stock_id]}
                      </p>
                    )}

                    <button
                      type="button"
                      onClick={() => handleDecrease(item)}
                      disabled={busy === item.stock_id || out}
                      className="w-full h-10 bg-[var(--color-primary)] hover:bg-[var(--color-primary-dark)] disabled:opacity-50 text-white text-sm font-semibold rounded-xl transition-colors cursor-pointer"
                    >
                      {busy === item.stock_id ? t("common.updating") : t("station.confirmTake")}
                    </button>
                  </div>
                )}
              </div>
            )
          })}
        </div>
      )}
    </div>
  )
}
```

### 5. `frontend/src/components/shared/Modal.jsx`
Adds `Escape` closing, scroll lock, and `aria-label` across all modals & `ConfirmModal`.

```jsx
import { useEffect } from "react"
import { cn } from "@/lib/utils"
import { X } from "lucide-react"

export function Modal({ open, onClose, title, children, maxWidth = "max-w-sm" }) {
  useEffect(() => {
    if (!open) return
    const prevOverflow = document.body.style.overflow
    document.body.style.overflow = "hidden"

    const onKeyDown = (e) => {
      if (e.key === "Escape" && onClose) onClose()
    }
    window.addEventListener("keydown", onKeyDown)
    return () => {
      document.body.style.overflow = prevOverflow
      window.removeEventListener("keydown", onKeyDown)
    }
  }, [open, onClose])

  if (!open) return null

  return (
    <div
      className="fixed inset-0 z-50 flex items-center justify-center bg-black/45 backdrop-blur-[2px] p-4"
      onClick={() => onClose?.()}
    >
      <div
        className={cn(
          "bg-[var(--color-surface)] rounded-2xl shadow-2xl w-full border border-[var(--color-border)] overflow-hidden",
          maxWidth
        )}
        onClick={(e) => e.stopPropagation()}
      >
        {title && (
          <div className="flex items-center justify-between px-5 py-3.5 border-b border-[var(--color-border)]">
            <div className="font-semibold text-sm text-[var(--color-text)]">{title}</div>
            {onClose && (
              <button
                type="button"
                onClick={onClose}
                aria-label="Close modal"
                className="p-1 rounded-lg text-[var(--color-muted)] hover:bg-[var(--color-surface-subtle)] hover:text-[var(--color-text)] transition-colors cursor-pointer"
              >
                <X size={16} />
              </button>
            )}
          </div>
        )}
        <div className="p-5">{children}</div>
      </div>
    </div>
  )
}
```
