Here is **Phase 3: Cashier Split-Screen POS, Live Order Board, Receipt Scanner & Payment Checkout**.

### What's improved in Phase 3
- **True Split-Screen POS (`CashierMenu.jsx`, `SubmitOrderDialog.jsx`, `CashierMenuGrid.jsx`):**
  - Desktop/tablet (`lg+`) gets a real 2-column POS workspace: **searchable menu + one-tap category pills** on the left, **sticky live order ticket** (table selector, occupied warning, guests, notes, qty steppers, USD/KHR totals, submit) on the right.
  - Mobile/small tablets keep the fixed dark-mode-safe `CartBar` (with proper `pb-28` so cards are never covered) + modal checkout.
  - Fixes silent ignoring of `party_size` / `note` on occupied tables, filters missing items to prevent `$NaN`, and uses `Promise.allSettled` so partial failures never duplicate items on retry.
- **POS Order Card Board (`CashierOrders.jsx`, `ActiveOrdersTable.jsx`, `ViewOrderDialog.jsx`):**
  - Replaces 3 stacked HTML tables with clean filterable POS order cards showing queue #, table, elapsed time, item preview, allergy highlights, USD/KHR total, and large touch-friendly **Pay / Print / Edit / Delete** actions.
  - **Live-synced `activePayOrder`** (fixes stale totals when items change while Payment modal is open).
  - **280ms debounced socket refresh** (turns a 6-event multi-item order burst into 1 API request).
- **React 19-Safe `ReceiptScanner.jsx`:** Replaced the `async` callback ref with a proper `useEffect` + cancellation cleanup so the camera never leaks, plus `Escape` key support.
- **Upgraded `PaymentDialog.jsx`:** Resets state cleanly per order, detects if the order total changed while a KHQR code is on screen (preventing wrong-amount scans), Escape + scroll lock, and cleaner bilingual labels.
- **Reliable Print Cleanup (`orderHelpers.js`):** Reuses shared `@/lib/currency` helpers and cleans up print iframes via `onafterprint` (with 60s fallback) instead of a risky 3s hard timeout.

```js
function generateBarcodeSvg(value) {
  try {
    const svg = document.createElementNS("http://www.w3.org/2000/svg", "svg")
    JsBarcode(svg, value, {
      format: "CODE128",
      width: 1.5,
      height: 36,
      displayValue: true,
      fontSize: 11,
      fontOptions: "bold",
      margin: 2,
      lineColor: "#000000",
      background: "#ffffff",
    })
    return svg.outerHTML
  } catch {
    return `<div style="font-family:monospace;font-size:12px;font-weight:bold;letter-spacing:2px;">${escapeHtml(value)}</div>`
  }
}

function scheduleIframePrint(iframe, beforePrint) {
  let cleaned = false
  const cleanup = () => {
    if (cleaned) return
    cleaned = true
    try {
      iframe.remove()
    } catch {}
  }

  setTimeout(() => {
    try {
      beforePrint?.()
      const win = iframe.contentWindow
      if (!win) {
        cleanup()
        return
      }
      win.onafterprint = cleanup
      win.focus()
      win.print()
      // Fallback cleanup for browsers that don't fire onafterprint reliably
      setTimeout(cleanup, 60000)
    } catch {
      cleanup()
    }
  }, 150)
}

export function printOrderReceipt(order) {
  if (!order) return

  const totalUSD = orderTotal(order)
  const totalKHR = Math.round(totalUSD * KHR_RATE)
  const orderCode = `ORD-${String(order.order_id).padStart(4, "0")}`
  const isPaid = (order.is_paid ?? order.status === "Paid") || order.payments?.length > 0
  const paymentMethod = order.payments?.[0]?.method?.method_name || (isPaid ? "Paid" : "Unpaid")
  const dateStr = order.order_date
    ? new Date(order.order_date).toLocaleString("en-GB")
    : new Date().toLocaleString("en-GB")
  const barcodeSvg = generateBarcodeSvg(orderCode)

  const itemsHtml = (order.order_items || [])
    .map((oi) => {
      const enName = oi.menu_item?.item_name || `Item #${oi.menu_item_id}`
      const kmName = oi.menu_item?.item_name_km
      const unitPrice = Number(oi.unit_price || 0)
      const lineTotal = unitPrice * Number(oi.quantity || 1)
      return `
        <tr>
          <td style="padding-right:4px;">
            <div class="bold">${escapeHtml(enName)}</div>
            ${kmName ? `<div class="item-km">${escapeHtml(kmName)}</div>` : ""}
            ${oi.note ? `<div class="item-note">* ${escapeHtml(oi.note)}</div>` : ""}

  const html = `

export function printDailyReport({ date, displayDate, tableLabel, data }) {
  if (!data) return
  const orders = data.orders || []
  const totalOrders = data.summary?.total_orders ?? orders.length
  const paidOrdersCount =
    data.summary?.paid_orders ?? orders.filter((o) => o.status === "Paid").length
  const totalRevenueUSD = Number(
    data.summary?.total_revenue ??
      orders.filter((o) => o.status === "Paid").reduce((s, o) => s + orderTotal(o), 0)
  )
  const totalRevenueKHR = Math.round(totalRevenueUSD * KHR_RATE)

  const methodMap = new Map()
  for (const o of orders.filter((x) => x.status === "Paid")) {
    const mName = o.payments?.[0]?.method?.method_name || "Cash / Other"
    const prev = methodMap.get(mName) || { count: 0, total: 0 }
    prev.count += 1
    prev.total += orderTotal(o)
    methodMap.set(mName, prev)
  }

  const methodsHtml = Array.from(methodMap.entries())
    .map(
      ([name, info]) => `
      <div class="meta-row">
        <span>${escapeHtml(name)} (${info.count})</span>
        <span class="bold">$${info.total.toFixed(2)}</span>
      </div>
    `
    )
    .join("")

  const orderRowsHtml = orders
    .map((o) => {
      const t = orderTotal(o)
      const time = new Date(o.order_date).toLocaleTimeString([], {
        hour: "2-digit",
        minute: "2-digit",
      })
      return `
        <tr>
          <td>#${String(o.order_id).padStart(4, "0")}</td>
          <td class="center">${escapeHtml(o.table?.table_number || "—")}</td>
          <td class="center">${escapeHtml(time)}</td>
          <td class="right bold">$${t.toFixed(2)}</td>
        </tr>
      `
    })
    .join("")

  const html = `
    <div class="center">
      <div class="brand-box">
        <div class="brand-title">ZOOM</div>
        <div class="brand-sub">DAILY CLOSING SLIP (Z-REPORT)</div>
      </div>
      <div style="font-size:10px;" class="bold">${escapeHtml(displayDate || date)}</div>
      ${tableLabel ? `<div style="font-size:10px;">Table: ${escapeHtml(tableLabel)}</div>` : ""}
    </div>

    <div class="divider-solid"></div>

    <div class="meta-row"><span>Total Orders:</span><span class="bold">${totalOrders}</span></div>
    <div class="meta-row"><span>Paid Orders:</span><span class="bold">${paidOrdersCount}</span></div>
    <div class="meta-row"><span>Unpaid / Open:</span><span class="bold">${totalOrders - paidOrdersCount}</span></div>

    ${methodsHtml
      ? `<div class="divider"></div>
         <div class="bold" style="font-size:10px; margin-bottom:2px;">BY PAYMENT METHOD:</div>
         ${methodsHtml}`
      : ""
    }

    <div class="divider-solid"></div>

    <div class="total-row">
      <span>REVENUE (USD)</span>
      <span>$${totalRevenueUSD.toFixed(2)}</span>
    </div>
    <div class="khr-row">
      <span>ចំណូលសរុប (KHR)</span>
      <span>៛${totalRevenueKHR.toLocaleString()}</span>
    </div>

    <div class="divider"></div>

    <table>
      <thead>
        <tr>
          <th>Order</th>
          <th class="center">Tbl</th>
          <th class="center">Time</th>
          <th class="right">Amount</th>
        </tr>
      </thead>
      <tbody>${orderRowsHtml}</tbody>
    </table>

    <div class="divider-solid"></div>

    <div style="margin-top:12px; font-size:10px;">
      <div>Cashier Sign: _______________________</div>
      <div style="margin-top:4px; font-size:9px;">Printed: ${escapeHtml(new Date().toLocaleString("en-GB"))}</div>
    </div>
  `

  printThermalReceiptHtml(`Daily Report - ${date}`, html)
}
```

### 2. `frontend/src/components/cashier/ReceiptScanner.jsx`
React 19-safe `useEffect` lifecycle (no `async` ref callback, guaranteed camera cleanup, `Escape` support).

```jsx
import { useEffect, useRef, useState, useCallback } from "react"
import { Html5Qrcode, Html5QrcodeSupportedFormats } from "html5-qrcode"
import { QrCode, X, Loader2 } from "lucide-react"
import { useLang } from "@/i18n/LanguageContext"

const SCANNER_ELEMENT_ID = "receipt-qr-scanner"

function extractOrderId(decodedText) {
  if (!decodedText) return null
  const clean = String(decodedText).trim()

  const prefMatch = clean.match(/(?:order:?|ORD-?|#)(\d+)/i)
  if (prefMatch) return parseInt(prefMatch[1], 10)

  const urlMatch = clean.match(/[?&/]order[=_/-]?(\d+)/i)
  if (urlMatch) return parseInt(urlMatch[1], 10)

  if (/^\d+$/.test(clean)) return parseInt(clean, 10)
  return null
}

export default function ReceiptScanner({ onFound, onClose }) {
  const { t, lang } = useLang()
  const [error, setError] = useState("")
  const scannerRef = useRef(null)
  const onFoundRef = useRef(onFound)
  onFoundRef.current = onFound

  const stopScanner = useCallback(async () => {
    const qr = scannerRef.current
    if (!qr) return
    scannerRef.current = null
    try {
      await qr.stop()
    } catch {}
    try {
      await qr.clear()
    } catch {}
  }, [])

  const handleClose = useCallback(async () => {
    await stopScanner()
    onClose()
  }, [stopScanner, onClose])

  useEffect(() => {
    let cancelled = false
    const prevOverflow = document.body.style.overflow
    document.body.style.overflow = "hidden"

    const onKeyDown = (e) => {
      if (e.key === "Escape") handleClose()
    }
    window.addEventListener("keydown", onKeyDown)

    const start = async () => {
      try {
        const qr = new Html5Qrcode(SCANNER_ELEMENT_ID, {
          formatsToSupport: [
            Html5QrcodeSupportedFormats.QR_CODE,
            Html5QrcodeSupportedFormats.CODE_128,
            Html5QrcodeSupportedFormats.CODE_39,
            Html5QrcodeSupportedFormats.EAN_13,
            Html5QrcodeSupportedFormats.UPC_A,
          ],
          verbose: false,
        })
        scannerRef.current = qr

        await qr.start(
          { facingMode: "environment" },
          { fps: 12, qrbox: { width: 260, height: 180 } },
          async (decodedText) => {
            if (cancelled) return
            const orderId = extractOrderId(decodedText)
            if (orderId) {
              cancelled = true
              await stopScanner()
              onFoundRef.current?.(orderId)
            } else {
              setError(
                lang === "km"
                  ? `បានស្កេន "${decodedText}" — សូមស្កេនបាកូដ ឬ QR លើវិក្កយបត្រ។`
                  : `Scanned "${decodedText}". Please scan the receipt barcode or QR code.`
              )
            }
          },
          () => {}
        )

        if (cancelled) {
          await stopScanner()
        }
      } catch (err) {
        if (!cancelled) {
          console.error(err)
          setError(t("guest.cameraError"))
        }
      }
    }

    start()

    return () => {
      cancelled = true
      document.body.style.overflow = prevOverflow
      window.removeEventListener("keydown", onKeyDown)
      stopScanner()
    }
  }, [handleClose, stopScanner, t, lang])

  return (
    <div className="fixed inset-0 z-50 bg-black/95 flex flex-col">
      <div className="flex items-center justify-between p-4 max-w-2xl w-full mx-auto">
        <div className="flex items-center gap-2.5 text-white">
          <div className="w-8 h-8 rounded-lg bg-white/10 flex items-center justify-center">
            <QrCode size={16} />
          </div>
          <span className="text-sm font-semibold">{t("cashier.scanReceipt")}</span>
        </div>
        <button
          type="button"
          onClick={handleClose}
          aria-label="Close scanner"
          className="p-2 rounded-full bg-white/10 hover:bg-white/20 text-white transition-colors cursor-pointer"
        >
          <X size={18} />
        </button>
      </div>

      <div className="flex-1 flex items-center justify-center px-4">
        <div
          id={SCANNER_ELEMENT_ID}
          className="w-full max-w-sm rounded-2xl overflow-hidden ring-2 ring-white/20 shadow-2xl"
        />
      </div>

      <div className="p-6 text-center text-white/75 text-xs space-y-2">
        {error ? (
          <div className="text-red-300 font-medium">{error}</div>
        ) : (
          <div className="inline-flex items-center justify-center gap-2 px-4 py-2 rounded-full bg-white/10">
            <Loader2 size={13} className="animate-spin text-emerald-300" />
            <span>
              {lang === "km"
                ? "តម្រង់កាមេរ៉ាទៅកាន់បាកូដ ឬ QR កូដលើវិក្កយបត្រ"
                : "Point camera at the barcode or QR code on the receipt"}
            </span>
          </div>
        )}
      </div>
    </div>
  )
}
```

```jsx
  const [partySize, setPartySize] = useState("")

  const [existingTableOrder, setExistingTableOrder] = useState(null)

  const resetTimerRef = useRef(null)

    fetchTables().catch(() => {})

  useSocketEvent("table:updated", safeFetchTables)
  useSocketEvent("order:new", safeFetchTables)
  useSocketEvent("order:deleted", safeFetchTables)
  useSocketEvent("payment:completed", safeFetchTables)

  useEffect(() => {
    fetchItems().catch(() => {})
    fetchCategories().catch(() => {})
    fetchTables().catch(() => {})

    if (!tableId) {

    }

        if (!cancelled) setExistingTableOrder(ord || null)
      })
      .catch(() => {
        if (!cancelled) setExistingTableOrder(null)
      })
    return () => {
      cancelled = true
    }
  }, [tableId, tables, getActiveOrderByTable])

        return matchesCat && menuItemMatchesSearch(i, search, catName)
      }),
```

```jsx
function formatElapsed(dateStr, nowMs) {
  if (!dateStr) return ""
  const diffMin = Math.max(0, Math.floor((nowMs - new Date(dateStr).getTime()) / 60000))
  if (diffMin < 1) return "Just now"
  if (diffMin < 60) return `${diffMin}m ago`
  const hrs = Math.floor(diffMin / 60)
  const mins = diffMin % 60
  return mins > 0 ? `${hrs}h ${mins}m` : `${hrs}h`
}
```

### 9. `frontend/src/components/cashier/PaymentDialog.jsx`
Live total-change guard for KHQR, Escape/scroll lock, clean state reset per order, and crisp POS layout.

```jsx
import { useState, useEffect, useRef, useCallback } from "react"
import { Check, QrCode, Loader2, X, CreditCard, Printer, RefreshCw, AlertTriangle } from "lucide-react"

import { printOrderReceipt } from "./orderHelpers"
import { useLang } from "@/i18n/LanguageContext"

export default function PaymentDialog({ order, paymentMethods = [], onClose, onPaid }) {
  const { t, lang } = useLang()

  const [payMethodId, setPayMethodId] = useState(null)

  const [cashGiven, setCashGiven] = useState("")
  const [cashCur, setCashCur] = useState("USD")

  const [payMode, setPayMode] = useState("normal")
  const [khqrData, setKhqrData] = useState(null)
  const [khqrGeneratedTotal, setKhqrGeneratedTotal] = useState(null)
  const [khqrCurrency, setKhqrCurrency] = useState("KHR")
  const [khqrGenerating, setKhqrGenerating] = useState(false)
  const [khqrPaid, setKhqrPaid] = useState(false)
  const [khqrError, setKhqrError] = useState(null)
  const [khqrConfirming, setKhqrConfirming] = useState(false)
  const [khqrCheckingNow, setKhqrCheckingNow] = useState(false)
  const [khqrCountdown, setKhqrCountdown] = useState(600)

  const khqrPollRef = useRef(null)
  const khqrDelayRef = useRef(null)
  const khqrCountdownRef = useRef(null)
  const khqrSessionReqRef = useRef(0)
  const handledPaidRef = useRef(false)
  const checkingInFlightRef = useRef(false)
  const onCloseRef = useRef(onClose)
  const onPaidRef = useRef(onPaid)

  useEffect(() => {
    onCloseRef.current = onClose
    onPaidRef.current = onPaid
  }, [onClose, onPaid])

  const stopKhqrPolling = useCallback(() => {
    khqrSessionReqRef.current += 1
    if (khqrDelayRef.current) clearTimeout(khqrDelayRef.current)
    if (khqrPollRef.current) clearInterval(khqrPollRef.current)
    if (khqrCountdownRef.current) clearInterval(khqrCountdownRef.current)
    khqrDelayRef.current = null
    khqrPollRef.current = null
    khqrCountdownRef.current = null
  }, [])

  const finishKhqrPayment = useCallback(() => {
    if (handledPaidRef.current) return
    handledPaidRef.current = true
    stopKhqrPolling()
    setKhqrPaid(true)
    setKhqrError(null)
    onPaidRef.current?.()
  }, [stopKhqrPolling])

  // Reset full dialog state when opening a different order
  useEffect(() => {
    if (!order?.order_id) return
    stopKhqrPolling()
    handledPaidRef.current = false
    checkingInFlightRef.current = false
    setPayMethodId(
      paymentMethods.find((m) => !/khqr|aba|bakong/i.test(m.method_name))?.method_id ||
        paymentMethods[0]?.method_id ||
        null
    )
    setPaying(false)
    setErrorMsg("")
    setCashGiven("")
    setCashCur("USD")
    setPayMode("normal")
    setKhqrData(null)
    setKhqrGeneratedTotal(null)
    setKhqrGenerating(false)
    setKhqrPaid(false)
    setKhqrError(null)
    setKhqrConfirming(false)
    setKhqrCheckingNow(false)
  }, [order?.order_id, stopKhqrPolling])

  useEffect(() => {
    if (!payMethodId && paymentMethods.length > 0) {
      setPayMethodId(
        paymentMethods.find((m) => !/khqr|aba|bakong/i.test(m.method_name))?.method_id ||
          paymentMethods[0]?.method_id ||
          null
      )
    }
  }, [paymentMethods, payMethodId])

  // Lock body scroll and allow Escape to close when not processing
  useEffect(() => {
    if (!order?.order_id) return

    const onKeyDown = (e) => {
      if (e.key === "Escape" && !paying && !khqrConfirming) {
        stopKhqrPolling()
        onCloseRef.current?.()
      }
    }
    window.addEventListener("keydown", onKeyDown)
    return () => {
      document.body.style.overflow = prevOverflow
      window.removeEventListener("keydown", onKeyDown)
    }
  }, [order?.order_id, paying, khqrConfirming, stopKhqrPolling])

  useEffect(() => () => stopKhqrPolling(), [stopKhqrPolling])

  // Invalidate dynamic KHQR if the live order total changes before payment completes
  const liveTotal = order ? orderTotal(order) : 0
  useEffect(() => {
    if (
      payMode === "khqr" &&
      !khqrPaid &&
      khqrData &&
      !khqrData.is_static &&
      khqrGeneratedTotal != null &&
      Math.abs(liveTotal - khqrGeneratedTotal) > 0.001
    ) {
      stopKhqrPolling()
      setKhqrData(null)
      setKhqrError(
        lang === "km"
          ? "តម្លៃសរុបបានផ្លាស់ប្តូរ — សូមបង្កើតកូដ KHQR ថ្មីម្តងទៀត។"
          : "Order total changed — please regenerate KHQR for the updated amount."
      )
    }
  }, [liveTotal, payMode, khqrPaid, khqrData, khqrGeneratedTotal, stopKhqrPolling, lang])

  useSocketEvent("payment:completed", (data) => {
    if (order?.order_id && Number(data?.order_id) === Number(order.order_id)) {
      finishKhqrPayment()
    }
  })

  useSocketEvent("order:status_changed", (data) => {
    if (
      order?.order_id &&
      Number(data?.order_id) === Number(order.order_id) &&
      data?.status === "Paid"
    ) {
      finishKhqrPayment()
    }
  })

  if (!order) return null

  const total = liveTotal
  const totalKHR = Math.round(total * KHR_RATE)
  const itemCount =
    order.order_items?.reduce((s, oi) => s + Number(oi.quantity || 0), 0) || 0

  const khqrMethod = paymentMethods.find((m) => /khqr|aba|bakong/i.test(m.method_name))
  const normalMethods = paymentMethods.filter((m) => m.method_id !== khqrMethod?.method_id)

  const selectedMethod = paymentMethods.find((m) => m.method_id === payMethodId)
  const isCash = /cash|សាច់ប្រាក់/i.test(selectedMethod?.method_name || "")
  const cashInput = parseFloat(cashGiven) || 0
  const cashUSD = cashCur === "KHR" ? cashInput / KHR_RATE : cashInput
  const change = cashUSD - total
  const isShort = change < -0.005
  const totalInCur = cashCur === "KHR" ? totalKHR : total
  const fmt = (usd) => (cashCur === "KHR" ? formatKHR(usd) : formatUSD(usd))

  const handleClose = () => {
    stopKhqrPolling()
    onCloseRef.current?.()
  }

  const checkMd5Now = async (
    dataToCheck = khqrData,
    showFeedback = false,
    reqId = khqrSessionReqRef.current
  ) => {
    if (
      !dataToCheck?.md5 ||
      !dataToCheck?.khqr_token ||
      dataToCheck.is_static ||
      handledPaidRef.current
    ) {
      return
    }
    if (!showFeedback && checkingInFlightRef.current) return
    checkingInFlightRef.current = true
    if (showFeedback) {
      setKhqrCheckingNow(true)
      setKhqrError(null)
    }
    try {
      const check = await api.get(
        `/payments/khqr/check/${encodeURIComponent(dataToCheck.md5)}?order_id=${order.order_id}&token=${encodeURIComponent(dataToCheck.khqr_token)}`,
        { silent: !showFeedback }
      )
      if (reqId !== khqrSessionReqRef.current || handledPaidRef.current) return
      if (check.data?.paid) {
        finishKhqrPayment()
      } else if (check.data?.ip_blocked) {
        stopKhqrPolling()
        setKhqrError(
          "Auto-confirm blocked by Bakong API (403). Verify payment on your phone and click confirm."
        )
      } else if (check.data?.token_missing || check.data?.auth_error) {
        stopKhqrPolling()
        setKhqrError(check.data.reason || "Bakong token not configured or expired on server.")
      } else if (check.data?.rate_limited) {
        stopKhqrPolling()
        setKhqrError(`Bakong API: ${check.data.reason || "Daily request limit exceeded."}`)
      } else if (showFeedback) {
        setKhqrError(
          check.data?.reason
            ? `Bakong status: ${check.data.reason}`
            : lang === "km"
              ? "មិនទាន់រកឃើញការទូទាត់នៅលើ Bakong នៅឡើយទេ។ សូមព្យាយាមម្តងទៀតបន្តិចទៀត។"
              : "Payment not detected on Bakong yet. Please try again in a few seconds."
        )
      }
    } catch (err) {
      if (reqId !== khqrSessionReqRef.current || handledPaidRef.current) return
      const status = err?.response?.status
      if (status === 409 && err?.response?.data?.already_paid) {
        finishKhqrPayment()
      } else if (status === 400 || status === 403 || status === 404 || status === 409) {
        stopKhqrPolling()
        setKhqrError(
          err?.response?.data?.message || "KHQR verification stopped. Please generate a new QR code."
        )
      } else if (showFeedback) {
        setKhqrError(err?.response?.data?.message || "Could not check Bakong status.")
      }
    } finally {
      checkingInFlightRef.current = false
      if (showFeedback) setKhqrCheckingNow(false)
    }
  }

  const startKhqrSession = async (curMode = khqrCurrency) => {
    stopKhqrPolling()
    const reqId = khqrSessionReqRef.current
    setKhqrError(null)
    setKhqrPaid(false)
    setKhqrData(null)
    setKhqrGenerating(true)
    handledPaidRef.current = false
    checkingInFlightRef.current = false
    setKhqrCountdown(600)

    const snapshotTotal = total

    try {
      const res = await api.post("/payments/khqr/generate", {
        order_id: order.order_id,
        currency: curMode,
      })
      if (khqrSessionReqRef.current !== reqId || handledPaidRef.current) return

      const generated = res.data
      setKhqrData(generated)
      setKhqrGeneratedTotal(snapshotTotal)

      const remainingSec = generated.expires_at
        ? Math.max(1, Math.floor((generated.expires_at - Date.now()) / 1000))
        : 600
      setKhqrCountdown(remainingSec)

      if (!generated.is_static && curMode !== "STATIC" && generated.md5) {

        const MAX_AUTO_CHECKS = 145

        const pollCheck = () => {
          if (handledPaidRef.current || khqrSessionReqRef.current !== reqId || document.hidden) {
            return
          }
          attempts += 1
          if (attempts > MAX_AUTO_CHECKS) {
            if (khqrPollRef.current) clearInterval(khqrPollRef.current)
            khqrPollRef.current = null
            return
          }
          checkMd5Now(generated, false, reqId)
        }

        khqrDelayRef.current = setTimeout(() => {
          pollCheck()
          khqrPollRef.current = setInterval(pollCheck, 4000)
        }, 4000)
      }

      khqrCountdownRef.current = setInterval(() => {
        setKhqrCountdown((prev) => {
          if (prev <= 1) {
            stopKhqrPolling()
            setKhqrData(null)
            setKhqrError(
              lang === "km"
                ? "កូដ QR បានផុតកំណត់ហើយ។ សូមជ្រើសរើសរូបិយប័ណ្ណខាងលើដើម្បីបង្កើតថ្មី។"
                : "QR code expired. Select a currency above to generate a new one."
            )
            return 0
          }
          return prev - 1
        })
      }, 1000)
    } catch (err) {
      if (khqrSessionReqRef.current !== reqId) return
      if (err?.response?.status === 409 && err?.response?.data?.already_paid) {
        finishKhqrPayment()
      } else {
        setKhqrError(err?.response?.data?.message || "Failed to generate KHQR code.")
      }
    } finally {
      if (khqrSessionReqRef.current === reqId) {
        setKhqrGenerating(false)
      }
    }
  }

    setKhqrCurrency(newCur)
    startKhqrSession(newCur)
  }

      return

    }

      }

    } finally {
      setPaying(false)
    }

  )
```

```jsx
  const [deleting, setDeleting] = useState(false)
  const [deleteError, setDeleteError] = useState("")
  const [search, setSearch] = useState("")
  const [statusTab, setStatusTab] = useState("all") // "all" | "Served" | "Preparing" | "Pending"
  const [scannerOpen, setScannerOpen] = useState(false)
  const [scanError, setScanError] = useState("")
  const [nowMs, setNowMs] = useState(() => Date.now())

  const refetchTimerRef = useRef(null)

  useSocketRole("cashier")

  // Debounce bursts of order_item events into a single refresh
  const scheduleFetchOrders = useCallback(() => {
    if (refetchTimerRef.current) clearTimeout(refetchTimerRef.current)
    refetchTimerRef.current = setTimeout(() => {
      refetchTimerRef.current = null
      fetchOrders().catch(() => {})
    }, 280)
  }, [fetchOrders])

  useEffect(() => {
    fetchOrders().catch(() => {})
    fetchPaymentMethods().catch(() => {})
    const poll = setInterval(() => fetchOrders().catch(() => {}), 60000)
    const clock = setInterval(() => setNowMs(Date.now()), 30000)
    return () => {
      clearInterval(poll)
      clearInterval(clock)
      if (refetchTimerRef.current) clearTimeout(refetchTimerRef.current)
    }
  }, [fetchOrders, fetchPaymentMethods])

  useSocketEvent("order:new", scheduleFetchOrders)
  useSocketEvent("order:status_changed", scheduleFetchOrders)
  useSocketEvent("order:deleted", scheduleFetchOrders)
  useSocketEvent("order_item:new", scheduleFetchOrders)
  useSocketEvent("order_item:status_changed", scheduleFetchOrders)
  useSocketEvent("order_item:deleted", scheduleFetchOrders)
  useSocketEvent("order_item:updated", scheduleFetchOrders)
  useSocketEvent("order_item:note_changed", scheduleFetchOrders)
  useSocketEvent("payment:completed", scheduleFetchOrders)

  const activeOrders = useMemo(
    () => (orders || []).filter((o) => !(o.is_paid ?? o.status === "Paid")),
    [orders]
  )

  // Always derive both modal orders from live store so totals/items are never stale
  const activeViewOrder = useMemo(
    () => (viewOrderId ? orders.find((o) => o.order_id === viewOrderId) || null : null),
    [orders, viewOrderId]
  )
  const activePayOrder = useMemo(
    () => (payOrderId ? orders.find((o) => o.order_id === payOrderId) || null : null),
    [orders, payOrderId]
  )

  const requestDeleteOrder = (order) => {
    setDeleteError("")
    setDeleteTarget({ type: "order", order })
  }

  const requestDeleteItem = (order, item) => {
    setDeleteError("")
    setDeleteTarget({ type: "item", order, item })
  }

  const handleConfirmDelete = async () => {
    if (!deleteTarget) return
    setDeleting(true)
    setDeleteError("")
    try {
      if (deleteTarget.type === "order") {
        await deleteOrder(deleteTarget.order.order_id)
        if (viewOrderId === deleteTarget.order.order_id) setViewOrderId(null)
        if (payOrderId === deleteTarget.order.order_id) setPayOrderId(null)
      } else {
        const res = await deleteOrderItem(deleteTarget.item.order_item_id)
        if (res?.orderDeleted) {
          if (viewOrderId === deleteTarget.order.order_id) setViewOrderId(null)
          if (payOrderId === deleteTarget.order.order_id) setPayOrderId(null)
        }
      }
      await fetchOrders()
      setDeleteTarget(null)
    } catch (err) {
      setDeleteError(err?.response?.data?.message || t("common.failed"))
    } finally {
      setDeleting(false)
    }
  }

  const handleReceiptFound = (orderId) => {
    setScannerOpen(false)
    const code = `#${String(orderId).padStart(4, "0")}`
    const found = orders.find((o) => o.order_id === orderId)
    if (!found) {
      setScanError(
        lang === "km" ? `រកមិនឃើញការកុម្ម៉ង់ ${code} ទេ។` : `Order ${code} not found.`
      )
      return
    }
    if (found.is_paid ?? found.status === "Paid") {
      setScanError(
        lang === "km"
          ? `ការកុម្ម៉ង់ ${code} ត្រូវបានទូទាត់ប្រាក់រួចហើយ។`
          : `Order ${code} has already been paid.`
      )
      return
    }
    setScanError("")
    setPayOrderId(found.order_id)
  }

  const searchQuery = search.trim().toLowerCase()
  const searchDigits = searchQuery.replace(/\D/g, "")
  const searchAsNumber = searchDigits ? parseInt(searchDigits, 10) : null

  const searchedOrders = useMemo(() => {
    if (!searchQuery) return activeOrders
    return activeOrders.filter((o) => {
      const orderIdStr = String(o.order_id)
      const padded = orderIdStr.padStart(4, "0")
      if (
        (searchAsNumber !== null && o.order_id === searchAsNumber) ||
        orderIdStr.includes(searchQuery) ||
        padded.includes(searchQuery) ||
        `#${orderIdStr}`.includes(searchQuery) ||
        `#${padded}`.includes(searchQuery) ||
        `ord-${padded}`.includes(searchQuery) ||
        `ord-${orderIdStr}`.includes(searchQuery)
      ) {
        return true
      }
      if (searchAsNumber !== null && o.queue_number === searchAsNumber) return true
      const tableNum = String(o.table?.table_number ?? "").toLowerCase()
      if (tableNum && (tableNum.includes(searchQuery) || `table ${tableNum}`.includes(searchQuery))) {
        return true
      }
      if (o.note && o.note.toLowerCase().includes(searchQuery)) return true
      if (o.order_items?.some((oi) => menuItemMatchesSearch(oi.menu_item, searchQuery))) return true
      return false
    })
  }, [activeOrders, searchQuery, searchAsNumber])

  const handleSearchKeyDown = (e) => {
    if (e.key === "Enter" && searchQuery) {
      if (searchedOrders.length === 1) {
        setPayOrderId(searchedOrders[0].order_id)
      } else if (searchAsNumber !== null) {
        const exact = activeOrders.find((o) => o.order_id === searchAsNumber)
        if (exact) setPayOrderId(exact.order_id)
      }
    }
  }

  const stats = useMemo(
    () => ({
      active: activeOrders.length,
      pending: activeOrders.filter((o) => o.status === "Pending").length,
      cooking: activeOrders.filter((o) => o.status === "Preparing").length,
      ready: activeOrders.filter((o) => o.status === "Served").length,
      total: activeOrders.reduce((s, o) => s + orderTotal(o), 0),
    }),
    [activeOrders]
  )

  const grouped = useMemo(
    () => ({
      Served: searchedOrders.filter((o) => o.status === "Served"),
      Preparing: searchedOrders.filter((o) => o.status === "Preparing"),
      Pending: searchedOrders.filter((o) => o.status === "Pending"),
    }),
    [searchedOrders]
  )

  const sectionConfig = [
    {
      key: "Served",
      label: t("cashier.readyForPayment"),
      sub: `${grouped.Served.length} ${t("cashier.orders")}`,
      accent: "text-[var(--color-primary)]",
      dot: "bg-[var(--color-primary)]",
    },
    {
      key: "Preparing",
      label: t("cashier.beingPrepared"),
      sub: `${grouped.Preparing.length} ${t("cashier.ordersInKitchen")}`,
      accent: "text-[var(--color-warning)]",
      dot: "bg-[var(--color-warning)]",
    },
    {
      key: "Pending",
      label: t("cashier.newOrders"),
      sub: `${grouped.Pending.length} ${t("cashier.ordersWaiting")}`,
      accent: "text-[var(--color-text-secondary)]",
      dot: "bg-[var(--color-muted)]",
    },
  ].filter((sec) => statusTab === "all" || sec.key === statusTab)

  const visibleCount = sectionConfig.reduce((s, sec) => s + grouped[sec.key].length, 0)

  const deleteCode = deleteTarget?.order
    ? `#${String(deleteTarget.order.order_id).padStart(4, "0")}`
    : ""

  return (
    <div className="space-y-5">
      {/* Top Toolbar: Search + Scan + Live Unpaid Total */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
        <div className="flex items-center gap-2 flex-1">
          <div className="relative flex-1 max-w-md">
            <Search
              size={15}
              className="absolute left-3.5 top-1/2 -translate-y-1/2 text-[var(--color-muted)] pointer-events-none"
            />
            <input
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              onKeyDown={handleSearchKeyDown}
              placeholder={t("cashier.searchPlaceholder")}
              className="w-full h-10 pl-10 pr-9 rounded-xl border border-[var(--color-border)] bg-[var(--color-surface)] text-sm text-[var(--color-text)] placeholder:text-[var(--color-muted)]/75 outline-none focus:ring-2 focus:ring-[var(--color-primary)]/25 focus:border-[var(--color-primary)] transition-all"
            />
            {search && (
              <button
                type="button"
                onClick={() => setSearch("")}
                aria-label="Clear search"
                className="absolute right-2.5 top-1/2 -translate-y-1/2 p-1 rounded-full text-[var(--color-muted)] hover:bg-[var(--color-surface-subtle)] cursor-pointer"
              >
                <X size={14} />
              </button>
            )}
          </div>

          <button
            type="button"
            onClick={() => {
              setScanError("")
              setScannerOpen(true)
            }}
            className="flex items-center gap-2 h-10 px-3.5 rounded-xl bg-[var(--color-surface)] border border-[var(--color-border)] text-xs sm:text-sm font-semibold text-[var(--color-text-secondary)] hover:border-[var(--color-primary)] hover:text-[var(--color-primary)] transition-colors shrink-0 cursor-pointer"
          >
            <QrCode size={15} />
            <span className="hidden sm:inline">{t("cashier.scanReceipt")}</span>
          </button>
        </div>

        {/* Calm Unpaid Summary Pill */}
        <div className="inline-flex items-center gap-3 px-4 py-2 rounded-xl bg-[var(--color-surface)] border border-[var(--color-border)] self-start sm:self-auto">
          <span className="text-xs text-[var(--color-muted)]">
            {stats.active} {t("cashier.statActive")}
          </span>
          <span className="h-4 w-px bg-[var(--color-border)]" />
          <span className="text-sm font-bold text-[var(--color-primary)] tabular-nums">
            {formatUSD(stats.total)}
          </span>
        </div>
      </div>

      {scanError && (
        <div
          role="alert"
          className="flex items-center justify-between gap-2 text-xs sm:text-sm text-[var(--color-danger)] bg-[var(--color-danger-muted)] border border-[var(--color-danger)]/25 rounded-xl px-4 py-2.5"
        >
          <span>{scanError}</span>
          <button
            type="button"
            onClick={() => setScanError("")}
            aria-label="Dismiss scan error"
            className="p-1 hover:opacity-75 cursor-pointer"
          >
            <X size={14} />
          </button>
        </div>
      )}

      {/* Clean Interactive Status Filter Tabs */}
      <div className="flex items-center gap-2 overflow-x-auto no-scrollbar pb-1">
        {[
          { key: "all", label: t("common.all"), count: stats.active },
          { key: "Served", label: t("status.ready"), count: stats.ready, dot: "bg-[var(--color-primary)]" },
          { key: "Preparing", label: t("status.cooking"), count: stats.cooking, dot: "bg-[var(--color-warning)]" },
          { key: "Pending", label: t("status.pending"), count: stats.pending, dot: "bg-[var(--color-muted)]" },
        ].map(({ key, label, count, dot }) => {
          const active = statusTab === key
          return (
            <button
              key={key}
              type="button"
              onClick={() => setStatusTab(key)}
              className={cn(
                "shrink-0 inline-flex items-center gap-2 px-3.5 py-2 rounded-xl text-xs font-semibold border transition-all cursor-pointer",
                active
                  ? "bg-[var(--color-primary)] text-white border-[var(--color-primary)] shadow-2xs"
                  : "bg-[var(--color-surface)] text-[var(--color-text-secondary)] border-[var(--color-border)] hover:border-[var(--color-primary)]/40"
              )}
            >
              {dot && !active && <span className={cn("w-2 h-2 rounded-full", dot)} />}
              <span>{label}</span>
              <span
                className={cn(
                  "text-[11px] px-1.5 py-0.5 rounded-full font-bold tabular-nums",
                  active
                    ? "bg-white/20 text-white"
                    : "bg-[var(--color-surface-subtle)] text-[var(--color-text)]"
                )}
              >
                {count}
              </span>
            </button>
          )
        })}
      </div>

      {/* Orders Grid Sections */}
      {isLoading && activeOrders.length === 0 ? (
        <LoadingState text={t("cashier.loadingOrders")} />
      ) : activeOrders.length === 0 ? (
        <EmptyState icon={ShoppingBag} text={t("cashier.noActiveOrders")} />
      ) : visibleCount === 0 ? (
        <EmptyState
          icon={ShoppingBag}
          text={
            searchQuery
              ? `${t("cashier.noMatchingOrders")} "${searchQuery}".`
              : t("cashier.noActiveOrders")
          }
        />
      ) : (
        <div className="space-y-6">
          {sectionConfig.map(
            ({ key, label, sub, accent, dot }) =>
              grouped[key].length > 0 && (
                <section key={key} className="space-y-3">
                  <div className="flex items-center gap-2">
                    <span className={cn("w-2.5 h-2.5 rounded-full", dot)} />
                    <h2 className={cn("font-semibold text-sm", accent)}>{label}</h2>
                    <span className="text-xs text-[var(--color-muted)]">· {sub}</span>
                  </div>
                  <ActiveOrdersTable
                    orders={grouped[key]}
                    nowMs={nowMs}
                    onPay={(ord) => setPayOrderId(ord.order_id)}
                    onView={(ord) => setViewOrderId(ord.order_id)}
                    onDelete={requestDeleteOrder}
                  />
                </section>
              )
          )}
        </div>
      )}

      <ViewOrderDialog
        order={deleteTarget ? null : activeViewOrder}
        onClose={() => setViewOrderId(null)}
        onDeleteItem={requestDeleteItem}
        onUpdateItemQty={updateOrderItemQuantity}
        onUpdateItemNote={updateOrderItemNote}
      />

      <ConfirmModal
        open={!!deleteTarget}
        onClose={() => !deleting && setDeleteTarget(null)}
        onConfirm={handleConfirmDelete}
        busy={deleting}
        danger
        title={
          deleteTarget?.type === "order"
            ? lang === "km"
              ? `លុបការកុម្ម៉ង់ ${deleteCode}?`
              : `Delete Order ${deleteCode}?`
            : lang === "km"
              ? "ដកមុខម្ហូបចេញពីការកុម្ម៉ង់?"
              : "Remove Order Item?"
        }
        message={
          deleteTarget?.type === "order"
            ? lang === "km"
              ? `លុបការកុម្ម៉ង់មិនទាន់ទូទាត់ ${deleteCode} (តុ ${deleteTarget.order.table?.table_number || "—"})? មិនអាចត្រឡប់វិញបានទេ។`
              : `Delete unpaid Order ${deleteCode} (Table ${deleteTarget.order.table?.table_number || "—"})? This cannot be undone.`
            : deleteTarget?.item
              ? lang === "km"
                ? `ដក ${menuItemName(deleteTarget.item.menu_item, lang)} (×${deleteTarget.item.quantity}) ចេញពីការកុម្ម៉ង់ ${deleteCode}?`
                : `Remove ${menuItemName(deleteTarget.item.menu_item, lang)} (×${deleteTarget.item.quantity}) from Order ${deleteCode}?`
              : ""
        }
        confirmLabel={t("common.delete")}
        cancelLabel={t("common.cancel")}
        errorMsg={deleteError}
      />

      <PaymentDialog
        order={activePayOrder}
        paymentMethods={paymentMethods}
        onClose={() => setPayOrderId(null)}
        onPaid={fetchOrders}
      />

      {scannerOpen && (
        <ReceiptScanner
          onFound={handleReceiptFound}
          onClose={() => setScannerOpen(false)}
        />
      )}
    </div>
  )
}
```
